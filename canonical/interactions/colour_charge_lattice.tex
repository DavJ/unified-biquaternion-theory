\section{Triplet-sum integrality proposition}
\subsection*{Step 1 (audit): colour-antisymmetric singlet in $\CC\otimes\HH$}
The $\ZZ_2^3$ involution structure gives three colour sectors at the representation level,
and the antisymmetric singlet tensor is
\[
\Psi_{\mathrm{singlet}}=\varepsilon^{abc}\,\Phi_a\otimes\Phi_b\otimes\Phi_c.
\]
Inside $\mathrm{Mat}(2,\CC)\cong \CC\otimes\HH$, the determinant
\(
\det M=M_{11}M_{22}-M_{12}M_{21}
\)
is a rank-2 antisymmetrisation. Therefore the 3-index colour antisymmetrisation is not a direct
determinant in $\mathrm{Mat}(2,\CC)$ alone; it is realised only after introducing the colour triplet
fiber $\CC^3$ already fixed by the SU(3) representation sector.
This resolves the dimension mismatch stated in this blocker audit (3-colour antisymmetrisation versus
2-index matrix determinant) \textbf{[L0]} and removes the invalid direct $\det(\Theta)$ shortcut.

\subsection*{Step 3: from $Q\in\ZZ$ to $Y\in\frac13\ZZ$}
\begin{proposition}[Colour-triplet hypercharge divisibility]
\label{prop:triplet_sum}
Let \(\widehat Y:=Y/2\) be the hypercharge variable in
\(
Q=T_3+\widehat Y
\)
equivalent to \texttt{qed.tex} (\(Q=T_3+Y/2\)).
For a colour triplet with colour-independent \(\widehat y\), if each colour component has integer EM charge
(\(Q_r,Q_g,Q_b\in\ZZ\)) and \(T_{3,r}+T_{3,g}+T_{3,b}\in\ZZ\), then
\[
3\widehat y\in\ZZ.
\]
Hence in PDG normalisation (\(Y_{\mathrm{PDG}}=\widehat Y\)):
\[
Y_{\mathrm{PDG}}\in\tfrac13\ZZ.
\]
\end{proposition}

\begin{proof}
From \(Q_i=T_{3,i}+\widehat y\) with colour-independent \(\widehat y\),
\[
Q_r+Q_g+Q_b=(T_{3,r}+T_{3,g}+T_{3,b})+3\widehat y.
\]
By assumptions, the left side is integer and the \(T_3\)-sum is integer, so \(3\widehat y\in\ZZ\).
Therefore \(\widehat y\in\frac13\ZZ\), i.e. \(Y_{\mathrm{PDG}}\in\frac13\ZZ\).
\end{proof}

\label{thm:colour_lattice}
Given Theorem~\ref{thm:C2-target} and Proposition~\ref{prop:triplet_sum}, the colour-triplet

Applying Proposition~\ref{prop:triplet_sum} yields \(3\widehat y\in\ZZ\), hence

\end{proof}

\section{Anomaly-minimality route for C2 uniqueness}
Consider the family \(Y_Q=n/6\), \(n\in\ZZ_{>0}\), with
\[
Q_u=\frac12+\frac{n}{6},\qquad Q_d=-\frac12+\frac{n}{6}.
\]
The anomaly-minimality program asks whether the chiral cubic condition
\(\sum_L Q^3-\sum_R Q^3=0\) plus full SM multiplet consistency selects only
\(n=1\) (equivalently \(Y_Q=1/6\)).

Current status in this file: this route is formulated, but uniqueness from
anomaly constraints alone is not yet closed here as an independent theorem.

\subsection*{Numerical check of the \(Y_Q=n/6\) family}
Using one-generation SM charge assignments with
\[
Q_{uL}=Q_{uR}=\frac12+\frac{n}{6},\quad
Q_{dL}=Q_{dR}=-\frac12+\frac{n}{6},\quad
Q_{\nu L}=0,\quad Q_{eL}=Q_{eR}=-1,
\]
the cubic chiral anomaly quantity
\[
\Delta Q^3:=\sum_{\mathrm L}Q^3-\sum_{\mathrm R}Q^3
\]
is numerically \(\Delta Q^3=0\) for all tested \(n\ge 1\) (including
\(n=1,\dots,10\)). Therefore this constraint alone does \emph{not}
single out \(n=1\), and uniqueness of \(Y_Q=1/6\) remains open in this route.

\subsection*{Mixed gravitational-gauge anomaly test}

The mixed gravitational-gauge anomaly for one generation is
\[
\mathcal A_{\rm grav}=\sum_f Y_f,
\]
where the sum runs over left-handed fermion multiplets.  For the family
\[
Y_Q=\frac{n}{6},\qquad Y_L=-\frac12,
\]
the explicit one-generation contribution is
\[
\mathcal A_{\rm grav}(n)=6Y_Q+2Y_L=n-1.
\]
Hence
\[
\mathcal A_{\rm grav}(1)=0,\qquad
\mathcal A_{\rm grav}(n)\neq 0 \ \text{for}\ n\ge 2.
\]
So this mixed-anomaly check selects \(n=1\) uniquely within the tested
family \(Y_Q=n/6\).

\begin{theorem}[Hypercharge uniqueness via gravitational anomaly]
\label{thm:C2-uniqueness}
Consider the one-parameter family of hypercharge assignments
$Y_Q = n/6$ for $n \in \mathbb{N}^+$, consistent with
$Y \in \tfrac{1}{6}\mathbb{Z}$ [L1, Prop.~\ref{prop:Y-lattice}].
The mixed $[\mathrm{U}(1)_Y]^2 \mathrm{grav}$ anomaly coefficient is
\[
  \mathcal{A}_{\rm grav}(n)
  = \sum_{\rm left} Y_f - \sum_{\rm right} Y_f = n - 1.
\]
The anomaly-free condition $\mathcal{A}_{\rm grav}(n) = 0$
selects $n = 1$ uniquely within the tested family.
Hence $Y_Q = \tfrac{1}{6}$ is the unique anomaly-free hypercharge
assignment in the family $\{n/6 : n \in \mathbb{N}^+\}$.
\end{theorem}

\begin{proof}
Direct calculation:
Left-handed contributions: quark doublet $Y=n/6$ (mult.~6),
lepton doublet $Y=-1/2$ (mult.~2).
Right-handed: $Y_{uR}=n/6+1/2$, $Y_{dR}=n/6-1/2$, $Y_{eR}=-1$.
$\mathcal{A}_{\rm grav} = 6\cdot(n/6) + 2\cdot(-1/2)
- 3\cdot(n/6+1/2) - 3\cdot(n/6-1/2) - (-1)
= n - 1 + 0 = n-1$.
Setting $n=1$: $\mathcal{A}_{\rm grav}=0$. $\square$
\end{proof}

\begin{remark}[Scope of the family check]
Theorem~\ref{thm:C2-uniqueness} establishes uniqueness within the
tested family $Y_Q \in \{1/6, 2/6, 3/6, \ldots\}$.
It does not exclude assignments outside this family
(e.g.\ irrational or negative $Y_Q$).
The full uniqueness proof (Gap~C2 Step~1) requires showing that
only $Y_Q = n/6$-type assignments are consistent with the UBT
algebra $\mathbb{C}\otimes_\mathbb{R}\mathbb{H}$; this remains [OPEN].
Status: \textbf{[L1 family check]}, not \textbf{[L1 full uniqueness]}.
\end{remark}

\section{Dirac quantisation on $T^2=S^1_\psi\times S^1_\chi$ (candidate)}
\begin{lemma}[Dirac quantisation on $T^2$]
\label{lem:dirac-T2}
Consider $T^2 = S^1_\psi \times S^1_\chi$ with U(1)$_Y$
holonomy groups of periods $p$ and $q$ respectively.
The Dirac quantisation condition on $T^2$ requires
$Y_{\rm min} = 1/(pq)$.
For $p=3$ (colour sector, from SU(3)) and $q=2$
(weak isospin sector, from SU(2)$_L$):
$Y_{\rm min} = 1/6$.
\end{lemma}

\begin{remark}
Interpretation of $(p,q)$ in current UBT status:
$p=3$ is motivated by three colour sectors (SU(3)$_c$),
and $q=2$ by weak-doublet structure (SU(2)$_L$).
In this file this remains a \textbf{[MC/candidate]} bridge:
the lattice unit $1/6$ is structurally motivated, but a strict derivation
of $(p,q)$ from the full $\mathbb{C}\otimes\mathbb{H}$ closure is still OPEN.
\end{remark}
