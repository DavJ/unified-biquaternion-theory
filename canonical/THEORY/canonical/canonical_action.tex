\begin{abstract}
This document states the current canonical UBT action without introducing an
independent spacetime metric.  The metric is the normalized compact-$\psi$-fiber
Gram observable of the existing field $\Theta(q,\tau)$.  The exact total

\section{Derived metric}

For
\begin{equation}
 \Theta(x,\psi)=\Theta(q_{\rm GR}(x),x^0+i\psi)\in\mathbb B
\end{equation}
define
\begin{equation}
 g_{\mu\nu}[\Theta]
 =\frac1{\mathcal N_0}\frac1{2\pi R_\psi}
 \int_0^{2\pi R_\psi}
 H(\partial_\mu\Theta,\partial_\nu\Theta)\,d\psi,
 \qquad \mathcal N_0>0\text{ constant}.
 \label{eq:metric}
\end{equation}
Here $H$ is the fixed real symmetric non-degenerate bilinear on the selected
metric-generating sector of $\mathbb B$.

\section{Composite UBT action}

\begin{definition}[Pure-$\Theta$ action]
\begin{equation}
 \boxed{
 \widehat S[\Theta]
 =S_{\rm EH}[g[\Theta]]+S_{\rm int}[g[\Theta],\Theta],}
 \label{eq:action}
\end{equation}
where
\begin{equation}
 S_{\rm EH}[g]
 =\frac1{2\kappa}\int_M\sqrt{-g}\,R[g]d^4x
\end{equation}
and $S_{\rm int}$ contains the UBT kinetic, gauge, source and potential sectors,
including the compact-$\psi$ integration required by their definitions.

The raw biquaternionic product
\begin{equation}
 \mathcal Q_{\mu\nu}

may enter internal UBT invariants.  The present Einstein--Hilbert term depends
only on the real scalar metric in Eq.~\eqref{eq:metric}; no undefined
noncommutative determinant of $\mathcal Q$ is assumed.

\section{Exact variation}

Define the direct fixed-metric internal Euler--Lagrange profile by
\begin{equation}
 \delta_\Theta S_{\rm int}\big|_g
 =\int_M\sqrt{-g}\,
 \langle\mathscr F_\Theta,\delta\Theta\rangle_\psi\,d^4x,
\end{equation}
and define the Einstein residual
\begin{equation}
 \mathscr E^{\mu\nu}=G^{\mu\nu}-\kappa T^{\mu\nu}.
\end{equation}
The metric variation is
\begin{equation}
 \delta g_{\mu\nu}
 =\frac2{\mathcal N_0}
 \langle\partial_{(\mu}\Theta,
 \partial_{\nu)}\delta\Theta\rangle_\psi.
\end{equation}
After covariant integration by parts, stationarity of
Eq.~\eqref{eq:action} gives
\begin{equation}
 \boxed{
 \mathscr F_\Theta
 +\frac1{\kappa\mathcal N_0}
 \nabla_\nu\left(\mathscr E^{\mu\nu}\partial_\mu\Theta\right)=0.}
 \label{eq:exactEL}
\end{equation}
This is the canonical pure-$\Theta$ variational equation.  Its two terms must
not be set to zero separately without an additional projection or theorem.
