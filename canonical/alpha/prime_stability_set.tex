\section{Canonical Model (Locked)}
\label{sec:pss:canonical}
%═══════════════════════════════════════════════════════════════════════════════

\CANONICAL{}

\begin{lockbox}
\textbf{CANONICAL MODEL — LOCKED.}
The following definition is the unique canonical prime-stability model for
the Unified Biquaternion Theory.  It must not be modified.
\end{lockbox}

\subsection{Definition}
\label{ssec:pss:def}

The prime-stability effective potential is:
\begin{equation}
  \label{eq:pss:Vcanon}
  V(p) \;=\; p^2 - B(p)\,p\ln p,
\end{equation}
where the self-consistent coupling is:
\begin{equation}
  \label{eq:pss:Bcanon}
  B(p) \;=\; \frac{p+1}{3}.
\end{equation}

\begin{tcolorbox}[colback=green!4!white, colframe=green!50!black,
  title=Canonical invariants (all terms must not be altered)]
\begin{enumerate}
  \item The entropy factor is $p\ln p$.  No alternative entropy form may
        replace $p\ln p$ in the canonical model.
  \item The coupling is $B(p) = (p+1)/3$.  No alternative coefficient may
        replace this expression.
  \item The potential has a canonical minimum structure defined by the
        self-consistency condition (Definition~\ref{def:pss:stable}).
\end{enumerate}
\end{tcolorbox}

\subsection{Stability Definition}
\label{ssec:pss:stability_def}

\begin{definition}[Stable prime]\label{def:pss:stable}
Let $n_c(B)$ denote the unique positive solution of
\begin{equation}
  2n = B\bigl(\ln n + 1\bigr),
  \label{eq:pss:ncrit}
\end{equation}
i.e., the continuous minimum of $V(n, B) = n^2 - B\,n\ln n$ over $n > 0$.
For a prime $p \ge 3$, define the prime-gap stability boundaries:
\begin{align}
  \ell(p) &= \tfrac{1}{2}\bigl(p_{\mathrm{prev}} + p\bigr), \\
  u(p)    &= \tfrac{1}{2}\bigl(p + p_{\mathrm{next}}\bigr),
\end{align}
where $p_{\mathrm{prev}}$ and $p_{\mathrm{next}}$ are the nearest primes below and above $p$.
The prime $p$ is \textbf{stable} if:
\[
  \ell(p) \;<\; n_c\bigl(B(p)\bigr) \;<\; u(p).
\]
The \textbf{stability margin} of $p$ is:
\begin{equation}
  \mathrm{margin}(p)
  \;=\;
  \min\!\bigl(n_c(B(p)) - \ell(p),\;\; u(p) - n_c(B(p))\bigr).
  \label{eq:pss:margin}
\end{equation}
The prime $p = 2$ is included as a base case: it is the unique prime for
which $V(2) > 0$ (the potential is positive, signalling a different regime).
\end{definition}

The physical meaning of this criterion is self-consistency: a prime~$p$ is
stable if and only if $p$ is its own closest-prime attractor under the
coupling $B(p)$.  The continuous minimum $n_c(B(p))$ falls within the
prime-gap around $p$, so no neighbouring prime is a better candidate.

\subsection{Renormalized Stirling Interpolation (Canonical)}
\label{ssec:pss:Sren}

\DERIVED{} The renormalized entropy is:
\begin{equation}
  S_{\mathrm{ren}}(p)
  \;=\;
  \ln\Gamma(p+1) + p - \tfrac{1}{2}\ln(2\pi p).
  \label{eq:pss:Sren}
\end{equation}

\begin{resultbox}
\textbf{Role of $S_{\mathrm{ren}}$}: $S_{\mathrm{ren}}$ is a renormalized
Stirling interpolation of $p\ln p$, \textbf{not} a new entropy model.

By Stirling's formula, $\ln\Gamma(p+1) = p\ln p - p + \tfrac{1}{2}\ln(2\pi p) + O(1/p)$.
Substituting:
\[
  S_{\mathrm{ren}}(p)
  = \bigl[p\ln p - p + \tfrac{1}{2}\ln(2\pi p)\bigr]
    + p - \tfrac{1}{2}\ln(2\pi p) + O(1/p)
  = p\ln p + O(1/p).
\]
The sub-leading Stirling corrections cancel exactly, leaving $p\ln p$ as the
leading term.  $S_{\mathrm{ren}}$ recovers the canonical entropy with
corrections of order $1/p$.
\end{resultbox}

The renormalized potential is:
\begin{equation}
  V_{\mathrm{ren}}(p) \;=\; p^2 - B(p)\,S_{\mathrm{ren}}(p).
  \label{eq:pss:Vren}
\end{equation}
Numerical agreement with $V(p)$ for all $p \in S$ is confirmed in
Table~\ref{tab:pss:margins} and \texttt{reports/prime\_stability\_revalidation.md}.

\subsection{Non-Canonical Gamma Variant (Excluded)}
\label{ssec:pss:gamma_noncanon}

\EXCLUDED{} \NOTUSED{}

\begin{excludedbox}
\textbf{Non-canonical entropy variant — excluded from canonical model.}

Direct substitution $p\ln p \to \ln\Gamma(p+1)$ (without the Stirling
renormalization) defines a \emph{different} model:
\[
  V_{\Gamma}(p) \;=\; p^2 - B(p)\,\ln\Gamma(p+1).
\]
This model has a stability structure centered near $p \approx e^6 \approx 403$,
not near $p = 137$.  The resulting stable-prime cluster around $p \approx 400$
is a feature of the non-renormalized Gamma model and does \textbf{not} belong
to the canonical model.

\textbf{Warning}: Any result derived from $V_\Gamma$ (including primes near 400)
must not be presented in the canonical theory context.  The cluster at
$p \approx 400$ arises solely from the non-renormalized Gamma entropy and is
excluded by construction.
\end{excludedbox}

\section{Exclusion of the $\sim 400$ Cluster}
\label{sec:pss:cluster400}
%═══════════════════════════════════════════════════════════════════════════════

\begin{excludedbox}
The cluster of primes near $p \approx 400$ arises only from the
non-renormalized Gamma entropy $\ln\Gamma(p+1)$.  It is \textbf{excluded}
from the canonical model.

No statement suggesting primes near $p = 400$ belong to the canonical
prime-stability analysis may appear in any canonical document.  If such
statements exist in legacy files, they must be marked \NOTUSED{} or corrected
to refer to the Gamma-excluded variant.
\end{excludedbox}

Physical origin of the exclusion: the bare Gamma function $\ln\Gamma(p+1) = \ln(p!)$
grows as $p\ln p - p + O(\ln p)$.  For the potential
$V_\Gamma(p) = p^2 - B(p)\ln(p!)$, the balance point between $p^2$ and
$B(p)\ln(p!)$ occurs at $p \approx e^6 \approx 403$, not at 137.
The renormalization $S_{\mathrm{ren}} = \ln\Gamma(p+1) + p - \tfrac{1}{2}\ln(2\pi p)$
subtracts the sub-leading terms and restores the canonical $p\ln p$ form,
which has its minimum structure at 137.
