\section{Central metric identity}
Let
\[
 E_\mu=\mathcal N_0^{-1/2}D_\mu\Theta
 =i e_\mu{}^0\mathbf1+e_\mu{}^k\mathbf e_k
\]
on the real Lorentz slice.  Quaternion conjugation gives
\[
 \boxed{\frac12(E_\mu^\sharp E_\nu+E_\nu^\sharp E_\mu)
 =g_{\mu\nu}\mathbf1},
\qquad
 g_{\mu\nu}=e_\mu{}^ae_\nu{}^b\eta_{ab}.
\]
The vector part cancels algebraically.  No trace, componentwise real part,
phase choice, or compact-fiber average occurs.

\section{Rank and Lorentz gauge}
For a nondegenerate tetrad, the differential of
$e\mapsto e\eta e^{\mathsf T}$ is surjective onto symmetric tensors.  Given
arbitrary $h_{\mu\nu}=h_{\nu\mu}$, choose
\[
 \delta e_\mu{}^a=\tfrac12 h_{\mu\rho}e^{\rho a};
\]
then $\delta g_{\mu\nu}=h_{\mu\nu}$.  Thus the rank is ten.  The kernel is the
six-dimensional local Lorentz algebra.

\section{Christoffel symbols and frame connection}
The Levi--Civita coordinate connection is
\[
 \mathring\Gamma^\rho{}_{\mu\nu}
 =\tfrac12g^{\rho\sigma}
 (\partial_\mu g_{\nu\sigma}+\partial_\nu g_{\mu\sigma}
 -\partial_\sigma g_{\mu\nu}).
\]
The Lorentz-frame connection is a different object.  Tetrad compatibility is
\[
 \partial_\mu e_\nu{}^a
 -\Gamma^\rho{}_{\mu\nu}e_\rho{}^a
 +\omega_\mu{}^a{}_b e_\nu{}^b=0.
\]
It follows that
\[
 \omega_\mu{}^a{}_b
 =e_b{}^\nu
  (\Gamma^\rho{}_{\mu\nu}e_\rho{}^a-\partial_\mu e_\nu{}^a).
\]
Therefore $\Gamma$ is not the real part of $\Omega$.  The object
$\Omega_\mu=\rho_*(\omega_\mu)$ is a representation of the frame connection
on the field space.

\section{Torsion and contorsion}
With
\[
 T^a=de^a+\omega^a{}_b\wedge e^b,
\]
a specified tetrad and torsion uniquely determine
\[
 \boxed{\omega=\mathring\omega(e)+K(T)},
 \qquad
 K_{abc}=\tfrac12(T_{cab}-T_{abc}-T_{bca}).
\]
The classical torsion-free GR branch has $K=0$.  This is a kinematic
reconstruction theorem; it does not derive the torsion equation from UBT.

\section{Flat representer and curved obstruction}
Every constant Lorentz tetrad has
\[
 \Theta_{\rm aff}=\Theta_0+\sqrt{\mathcal N_0}\,\bar E_\mu x^\mu.
\]
For $\bar E_0=i\mathbf1$, $\bar E_k=\mathbf e_k$, this gives Minkowski
spacetime in an inertial gauge with $\Gamma=\omega=\Omega=0$.

A naive one-sided derivative with invertible $\Theta$ is incompatible with
generic nonzero curvature under torsion-free antisymmetric tetrad
compatibility.  The two-sided derivative
\[
 D_\mu\Theta=\partial_\mu\Theta+A_\mu\Theta-\Theta B_\mu
\]
instead obeys
\[
 [D_\mu,D_\nu]\Theta
 =F^A_{\mu\nu}\Theta-\Theta F^B_{\mu\nu},
\]
so nonzero curvatures may be intertwined by $\Theta$.

\section{Conditional dynamical subclosures and remaining work}

\[
 E_\mu=\mathcal N_0^{-1/2}
