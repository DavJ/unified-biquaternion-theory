requires separation of the direct internal $\Theta$ Euler--Lagrange equation from
the metric-induced equation; this final single-action separation is stated

\section{Foundational objects and notation}

Let $M$ be a smooth four-dimensional manifold with local coordinates
$x^\mu$, $\mu=0,1,2,3$.  The canonical complex time is
\begin{equation}
  \tau=x^0+i\psi,
  \qquad \psi\in S^1_{R_\psi},
  \qquad L_\psi=2\pi R_\psi.
\end{equation}
The fundamental field is the existing UBT field
\begin{equation}
  \Theta=\Theta(q_{\rm GR}(x),x^0+i\psi)\in\B
  =\mathbb C\otimes_{\mathbb R}\mathbb H,
  \qquad \dim_{\mathbb R}\B=8.
\end{equation}
There is no additional chronofactor field: $\psi$ is the imaginary part of the
same $\tau$ that already appears in $\Theta(q,\tau)$.

Fix a real, symmetric, non-degenerate bilinear form $H$ on the selected real
metric-generating representation of $\B$.  In a matrix representation it may be
written
\begin{equation}
  H(A,B)=\operatorname{Re}\operatorname{Tr}
  \bigl(A^\ddagger\eta_{\B}B\bigr),
  \label{eq:H}
\end{equation}
where $\ddagger$ is the full Hermitian involution and $\eta_{\B}$ is the chosen
representation-level Lorentzian form.  The closure proof requires only that
$H$ is real, symmetric and non-degenerate; it does not assume a particular
matrix realization.

\begin{definition}[Fiber pairing]
For two $\B$-valued profiles $U(\psi),V(\psi)$ define
\begin{equation}
  \langle U,V\rangle_\psi
  :=\frac1{L_\psi}\int_0^{L_\psi}H(U(\psi),V(\psi))\,\dd\psi.
  \label{eq:fiberpair}
\end{equation}
The completion of the smooth periodic profiles with this pairing is denoted
$\mathscr V_\psi$.  It is an infinite-dimensional real Krein space whenever
$H$ is indefinite.
\end{definition}

\section{The raw biquaternionic quadratic tensor and the physical metric}

\begin{definition}[Raw quadratic tensor]
Set
\begin{equation}
  \mathcal Q_{\mu\nu}(x,\psi)
  :=(\partial_\mu\Theta)(\partial_\nu\Theta)^\ddagger\in\B.
  \label{eq:rawQ}
\end{equation}
It obeys $\mathcal Q_{\nu\mu}=\mathcal Q_{\mu\nu}^\ddagger$.
\end{definition}

\begin{proposition}[Scalar-part correction]
For $A,B\in\B$, both $\Sc(AB^\ddagger)$ and the matrix trace of a faithful
$2\times2$ complex representation are complex scalars.  Consequently,
\begin{equation}
  \Sc(AB^\ddagger)\in\mathbb C,
  \qquad
  \operatorname{Tr}(AB^\ddagger)\in\mathbb C,
\end{equation}
not general elements of $\B$.
\end{proposition}

\begin{proof}
By definition, $\Sc:\B\to\mathbb C$ extracts the coefficient of the quaternion
identity.  The ordinary matrix trace maps $\operatorname{Mat}(2,\mathbb C)$ to
$\mathbb C$.  Neither operation can retain the three quaternion-vector
components.  The full biquaternionic information is carried by
$\mathcal Q_{\mu\nu}$ before scalar projection.
\end{proof}

\begin{remark}
This distinction repairs an ambiguity in older UBT texts that called
$\Sc(E_\mu E_\nu^\ddagger)$ itself a ``biquaternionic metric.''  It is more
precise to call $\mathcal Q_{\mu\nu}$ the raw biquaternionic quadratic tensor and
its scalar/real projection the metric channel.
\end{remark}

  \label{eq:metric}

The constant $\Nzero$ is fixed once by asymptotic Minkowski normalization.
It is not a local functional of $\partial_0\Theta$.

\begin{theorem}[Uniqueness of the compact-fiber zero-mode projection]
Let $F_{\mu\nu}(\psi):=H(\partial_\mu\Theta(\psi),
\partial_\nu\Theta(\psi))$ be the real quadratic metric channel on the compact
circle.  Suppose the classical readout $\Pi(F)$ is real-linear, continuous on
trigonometric polynomials (and hence on its chosen completion), normalized by
$\Pi(1)=1$, and invariant under every fiber translation
$T_aF(\psi)=F(\psi+a)$.  Then
\begin{equation}
 \Pi(F)=\frac1{L_\psi}\int_0^{L_\psi}F(\psi)\,d\psi.
 \label{eq:haarprojection}
\end{equation}
Thus Eq.~\eqref{eq:metric} is the unique normalized linear quadratic readout
that does not select a preferred point of the canonical compact
$S^1_\psi$ fiber.
\end{theorem}

\begin{proof}
Write a trigonometric polynomial as
$F(\psi)=\sum_{n=-N}^{N}F_n e^{in\psi/R_\psi}$.  Translation invariance gives
\begin{equation}
 \Pi(e^{in(\psi+a)/R_\psi})=\Pi(e^{in\psi/R_\psi})
\end{equation}
for every $a$.  For $n\ne0$, choose $a$ with
$e^{ina/R_\psi}\ne1$; linearity then forces
$\Pi(e^{in\psi/R_\psi})=0$.  Normalization gives $\Pi(1)=1$, so
$\Pi(F)=F_0$, which is precisely the normalized circle average.  Continuity
extends the result to the selected profile completion.
\end{proof}

\begin{remark}[Scope of the uniqueness theorem]
The theorem derives the fiber average from compact-fiber translation symmetry,
linearity and the already chosen quadratic metric channel.  If UBT instead
postulates that a physical observer selects a distinguished value of $\psi$,
then the theorem does not apply; that point-section theory is the rank-deficient
branch analysed below.
\end{remark}

\section{Exact first variation of the induced metric}

Write
\begin{equation}
  E_\mu:=\partial_\mu\Theta\in\mathscr V_\psi.
\end{equation}
For a compactly supported variation $\delta\Theta$ commuting with the fiber
integral,
\begin{equation}
  \boxed{
  \delta g_{\mu\nu}
  =\frac1{\Nzero}
  \bigl(
   \langle\partial_\mu\delta\Theta,E_\nu\rangle_\psi
  +\langle E_\mu,\partial_\nu\delta\Theta\rangle_\psi
  \bigr)
  =\frac2{\Nzero}
  \langle E_{(\mu},\partial_{\nu)}\delta\Theta\rangle_\psi.}
  \label{eq:deltag}
\end{equation}
This formula contains no independent variation of $g$.

\section{Pure-$\Theta$ action and its exact Euler--Lagrange equation}

Consider the canonical composite action
\begin{equation}
  \widehat S[\Theta]
  =S_{\rm EH}[g[\Theta]]+S_{\rm int}[g[\Theta],\Theta],
  \label{eq:action}
\end{equation}
where $S_{\rm int}$ denotes all direct matter, gauge and potential terms.
Define the fixed-metric direct Euler--Lagrange profile $\F_\Theta\in\mathscr
V_\psi^*$ by
\begin{equation}
 \delta_\Theta S_{\rm int}\big|_g
 =\int_M\sqrt{-g}\,\langle\F_\Theta,\delta\Theta\rangle_\psi\,\dd^4x,
 \label{eq:Ftheta}
\end{equation}
and define
\begin{equation}
  \E^{\mu\nu}:=G^{\mu\nu}-\kappa T^{\mu\nu}.
  \label{eq:Einres}
\end{equation}
With the convention
\begin{equation}
  \delta_g(S_{\rm EH}+S_{\rm int})
  =-\frac1{2\kappa}\int_M\sqrt{-g}\,
  \E^{\mu\nu}\delta g_{\mu\nu}\,\dd^4x,
\end{equation}
substitution of Eq.~\eqref{eq:deltag} gives
\begin{equation}
  \delta\widehat S
  =\int_M\sqrt{-g}\,\langle\F_\Theta,\delta\Theta\rangle_\psi\,\dd^4x
  -\frac1{\kappa\Nzero}\int_M\sqrt{-g}\,
   \E^{\mu\nu}\langle E_\mu,\partial_\nu\delta\Theta\rangle_\psi\,\dd^4x.
\end{equation}
After covariant integration by parts,
\begin{equation}
 \boxed{
 \F_\Theta
 +\frac1{\kappa\Nzero}
 \nabla_\nu\bigl(\E^{\mu\nu}E_\mu\bigr)=0.}
 \label{eq:fullEL}
\end{equation}
Equation~\eqref{eq:fullEL} is the exact pure-$\Theta$ variational equation for
the action~\eqref{eq:action}, under the stated fixed target pairing and constant
normalization.  It is not obtained by importing an external gravitational
formalism.

\begin{proposition}[Noether identity for the internal sector]
If $S_{\rm int}$ is diffeomorphism invariant and all gauge-field equations in
that sector hold, then
\begin{equation}
  \nabla_\mu T^{\mu}{}_{\nu}
  =-\langle\F_\Theta,E_\nu\rangle_\psi.
  \label{eq:noether}
\end{equation}
In particular, $\F_\Theta=0$ implies $\nabla_\mu T^{\mu\nu}=0$.
\end{proposition}

\begin{proof}
Use the infinitesimal diffeomorphism variation
$\delta_\xi\Theta=\xi^\nu\partial_\nu\Theta$ and
$\delta_\xi g_{\mu\nu}=2\nabla_{(\mu}\xi_{\nu)}$ in
$\delta S_{\rm int}=0$, integrate the stress-tensor term by parts, and use the
arbitrariness of $\xi^\nu$.
\end{proof}

\section{Normal closure operator derived from UBT}

\begin{equation}

\begin{equation}
  B_{\mu\nu}
  :=P_\perp\bigl(\nabla_\mu E_\nu\bigr)
  =B_{\nu\mu}\in\mathscr N_x.
  \label{eq:Bmunu}
\end{equation}
The symmetry follows locally from $E_\nu=\partial_\nu\Theta$ and the

\begin{equation}
  \boxed{\E^{\mu\nu}B_{\mu\nu}=0\quad\text{in }\mathscr N_x.}
  \label{eq:normalclosure}
\end{equation}
The tangential part vanishes identically; the physical content is the normal
closure equation~\eqref{eq:normalclosure}.

\begin{theorem}[Local pure-UBT vacuum GR closure]
For the pure gravitational action $S_{\rm EH}[g[\Theta]]$, let $\Theta$ be a

\begin{equation}
  G_{\mu\nu}=0
\end{equation}
pointwise.
\end{theorem}

\begin{proof}
For the pure gravitational action, $\F_\Theta=0$ and $T_{\mu\nu}=0$.

\begin{theorem}[Local matter/gauge GR closure on the canonical internal shell]
Let $\Theta$ be regular and fiber-free and suppose the fixed-metric internal

  \label{eq:internalEL}

$\mathcal K_x(\E)=0$.  Fiber-freeness makes $\mathcal K_x$ injective, hence
$\E^{\mu\nu}=0$.
\end{proof}

\section{Why one $\psi$-section cannot close general GR}

Suppose instead that the metric is built from a single value
$\Theta(x,\psi_0)\in\B\cong\mathbb R^8$.  At a regular point the tangent span has
real dimension four, so its orthogonal complement has dimension four.
Consequently all ten $B_{\mu\nu}$ lie in a four-dimensional normal space.

\begin{theorem}[Single-section rank no-go]
For a genuinely biquaternion-valued field with an eight-real-dimensional value
space, the single-section closure map satisfies
\begin{equation}
  \operatorname{rank}\mathcal K_x\le4,
  \qquad
  \dim\ker\mathcal K_x\ge 10-4=6.
\end{equation}
Therefore Eq.~\eqref{eq:normalclosure} cannot imply an arbitrary ten-component
Einstein residual vanishes pointwise.  Even after using four differential
Bianchi identities, the single-section construction does not provide a generic
algebraic proof of the six independent gravitational equations.
\end{theorem}

\begin{proof}
The image of $\mathcal K_x$ is contained in the four-dimensional normal space.
Rank--nullity gives the stated bound.  Bianchi identities are differential
identities and do not convert a rank-four pointwise map into an injective map on
the ten-dimensional symmetric-tensor fiber.
\end{proof}

\begin{corollary}[Holomorphy cannot repair the single-section rank]
Imposing holomorphy in $\tau$, a Jacobi heat equation, or any other constraint
on the same eight-real-valued field cannot increase the image of the
single-section metric-variation map.  Such constraints restrict admissible
variations and can only preserve or reduce its rank.
\end{corollary}

\section{The existing $\psi$-fiber removes the kinematic obstruction}

The fiber-completed field at fixed $x$ is the profile
$\psi\mapsto\Theta(x,\psi)$, an element of the infinite-dimensional space
$\mathscr V_\psi$.  Thus the normal space is no longer limited to four
real directions.

\begin{theorem}[Existence of holomorphic periodic fiber-free UBT jets]
There exist local configurations, analytic in the spatial coordinates and
holomorphic in $\tau=x^0+i\psi$, for which the metric is Lorentzian at a
chosen point and the ten vectors in Eq.~\eqref{eq:tenB} are linearly
independent.
\end{theorem}

\begin{proof}
Put
\begin{equation}
  \phi_m(\tau):=\exp(m\tau/R_\psi),\qquad m\in\mathbb Z.
\end{equation}
Each $\phi_m$ is holomorphic in $\tau$ and periodic under
$\psi\mapsto\psi+2\pi R_\psi$.  For the Hermitian real pairing in
Eq.~\eqref{eq:H}, profiles with distinct Fourier numbers are orthogonal after
fiber averaging.  Choose four mutually $H$-orthogonal algebraic directions
$u_0,u_1,u_2,u_3\in\B$ with $H(u_0,u_0)<0$ and
$H(u_i,u_i)>0$ for $i=1,2,3$.

Choose pairwise disjoint pairs of nonzero integers
$(a_0,b_0),(a_1,b_1),(a_2,b_2),(a_3,b_3)$, with $a_\mu\ne b_\mu$, and define
holomorphic profiles
\begin{align}
 f(\tau)&:=u_0\bigl(c_{a_0}\phi_{a_0}(\tau)
                   +c_{b_0}\phi_{b_0}(\tau)\bigr),\\
 g_i(\tau)&:=u_i\bigl(c_{a_i}\phi_{a_i}(\tau)
                   +c_{b_i}\phi_{b_i}(\tau)\bigr),\qquad i=1,2,3,
\end{align}
with both coefficients in every pair nonzero.  Use six additional Fourier
numbers, disjoint from all the preceding ones, to choose six mutually
orthogonal nonzero profiles $h_{ij}=h_{ji}$, $1\le i\le j\le3$.
Now set, near a point whose spatial coordinates are zero,
\begin{equation}
 \Theta(x,\psi)
 =f(\tau)+\sum_{i=1}^3x^i g_i(\tau)
  +\frac12\sum_{i,j=1}^3x^ix^j h_{ij}(\tau).
 \label{eq:holomorphicjet}
\end{equation}
This expression is holomorphic in $\tau$; no independent polynomial in
$x^0=\operatorname{Re}\tau$ has been introduced.

At the chosen point,
\begin{equation}
 E_0=f'(\tau),\qquad E_i=g_i(\tau),
\end{equation}
and the four tangent profiles are mutually orthogonal.  Rescaling the
coefficients fixes their norms to $(-,+,+,+)$, so the induced metric is
Lorentzian.  The normal second-derivative profiles are
\begin{equation}
 B_{00}=P_\perp f'',\qquad
 B_{0i}=P_\perp g_i',\qquad
 B_{ij}=h_{ij}.
\end{equation}
For a two-frequency profile $w=c_a\phi_a+c_b\phi_b$ with $a\ne b$ and
$c_ac_b\ne0$, its derivative is not proportional to $w$.  Therefore the
orthogonal component of $w'$ relative to $w$ is nonzero.  Applied to $f'$ and
$f''$, and to each pair $g_i,g_i'$, this proves that $B_{00}$ and all three
$B_{0i}$ are nonzero.  Their Fourier supports are pairwise disjoint, and the
six $h_{ij}$ have six further disjoint supports.  Hence all ten profiles
$B_{\mu\nu}$ are linearly independent.
\end{proof}

\begin{theorem}[Finite-mode genericity of fiber-freeness]
Fix a finite-dimensional holomorphic Fourier--Galerkin two-jet space that
contains at least the mutually disjoint mode pairs and six additional modes used
in Eq.~\eqref{eq:holomorphicjet}.  On its regular Lorentzian locus, the
fiber-free subset is nonempty and open dense (indeed the complement is contained
in the zero set of a nonzero polynomial minor).
\end{theorem}

\begin{proof}
Choose ten fixed continuous normal test functionals and form the corresponding
$10\times10$ closure minor.  On the regular locus its entries are rational
functions of the jet coefficients because $P_\perp$ contains the inverse
tangent Gram matrix.  After multiplication by a sufficient power of the
nonzero Gram determinant, the minor becomes polynomial.  The explicit
construction above makes the resulting polynomial nonzero, so its nonzero set
is open and dense in the ordinary finite-dimensional topology.  Intersecting
with the open regular-Lorentzian first-jet locus gives the claim.
\end{proof}

\begin{remark}[What this genericity theorem does and does not prove]
It proves nonemptiness and genericity in every sufficiently rich finite-mode
holomorphic truncation.  It does not establish density in a specified
infinite-dimensional Fr\'echet topology, nor that a particular fixed Jacobi
theta ansatz or every global solution of the master equation is fiber-free.
Those are separate functional-analytic and dynamical questions.
\end{remark}
