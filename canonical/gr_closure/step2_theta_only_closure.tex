\section{$\Theta$-Only Emergent Closure}
\label{sec:theta_only_closure}

\begin{tcolorbox}[colback=red!4!white,colframe=red!70!black,title=Supersession notice]
The earlier claim that gauge-reduced injectivity of $\Theta\mapsto g[\Theta]$
was enough to recover arbitrary metric stationarity was incorrect: injectivity
of the field-to-metric map does not imply surjectivity of its differential onto
the ten-dimensional metric-variation space.  The valid criterion is
injectivity of the normal closure operator defined below.
\end{tcolorbox}

The canonical field is the existing fiber-resolved field
\begin{equation}
 \Theta(x,\psi)=\Theta(q_{\rm GR}(x),x^0+i\psi)\in\mathbb B,
\end{equation}
and the physical metric is the normalized fiber average
\begin{equation}
 g_{\mu\nu}[\Theta]
 =\frac1{\mathcal N_0}\frac1{2\pi R_\psi}
 \int H(\partial_\mu\Theta,\partial_\nu\Theta)\,d\psi.
\end{equation}

For
\begin{equation}
 \widehat S[\Theta]=S_{\rm EH}[g[\Theta]]+S_{\rm int}[g[\Theta],\Theta],
\end{equation}
define
\begin{equation}
 \mathscr E^{\mu\nu}=G^{\mu\nu}-\kappa T^{\mu\nu}
\end{equation}
and let $\mathscr F_\Theta$ be the fixed-metric Euler--Lagrange derivative of
$S_{\rm int}$.  Direct variation gives
\begin{equation}
 \boxed{
 \mathscr F_\Theta
 +\frac1{\kappa\mathcal N_0}
 \nabla_\nu\bigl(\mathscr E^{\mu\nu}\partial_\mu\Theta\bigr)=0.}
 \label{eq:combined_EL_revised}
\end{equation}
No independent metric variation is used.

Let $B_{\mu\nu}$ be the normal projection of
$\nabla_\mu\partial_\nu\Theta$ in the real space of complete $\psi$ profiles.
On the direct internal shell $\mathscr F_\Theta=0$, diffeomorphism invariance
and the Bianchi identity reduce Eq.~\eqref{eq:combined_EL_revised} to
\begin{equation}
 \mathscr E^{\mu\nu}B_{\mu\nu}=0.
 \label{eq:normal_closure_revised}
\end{equation}

\begin{theorem}[Fiber-free local GR closure]
If the ten normal profile vectors $B_{\mu\nu}$, $\mu\le\nu$, are linearly
independent, then Eq.~\eqref{eq:normal_closure_revised} implies
\begin{equation}
 G_{\mu\nu}=\kappa T_{\mu\nu}.
\end{equation}
\end{theorem}

\begin{proof}
The map
$X^{\mu\nu}\mapsto X^{\mu\nu}B_{\mu\nu}$ from symmetric tensors to the normal
profile space is injective precisely under the stated linear-independence
condition.  Apply it to $X=\mathscr E$.
\end{proof}

\begin{proposition}[Single-section no-go]
At a fixed $\psi_0$, a genuinely biquaternion-valued field takes values in an
eight-real-dimensional space.  A regular four-dimensional tangent span then
has a four-dimensional normal complement, so the corresponding closure map has
rank at most four and cannot be injective on ten-component symmetric tensors.
\end{proposition}

The complete compact $\psi$ fiber avoids this dimensional obstruction because
the profile space is infinite-dimensional.  Fiber-free analytic two-jets exist and are open dense in every sufficiently rich finite-mode holomorphic truncation.  The full proof is
\begin{center}
\texttt{canonical/gr\_closure/pure\_ubt\_fiber\_closure.tex}.
\end{center}
