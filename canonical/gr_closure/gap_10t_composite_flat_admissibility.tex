\begin{abstract}
Consider the auxiliary gradient-composite restriction in which a Lorentz-real
field is written as $\Theta=Y^a\mathbf u_a$ and the tetrad is defined by
$e^a=\mathcal N_0^{-1/2}dY^a$.  We prove the stronger statement hidden behind
the previously reported affine-stationarity calculation: every nondegenerate
configuration of this restriction is locally a pullback of Minkowski space and
therefore has identically vanishing Riemann curvature.  The Hilbert--Palatini
term vanishes identically, while the locked kinetic and cosmological terms are
multiples of the Jacobian volume and are null Lagrangians under compactly
supported variations.  Hence the affine representer is stationary for all
coefficients, but only as a corollary of a flatness no-go.  This auxiliary
$\partial$-gradient restriction cannot be called the surviving GR branch of
canonical UBT, whose tetrad is $E_\mu=\mathcal N_0^{-1/2}D_\mu\Theta$ and whose
self-consistent $D$-composite variation remains open.
\end{abstract}

\section{Auxiliary gradient-composite restriction}
On the Lorentz slice write
\begin{equation}
 \Theta=Y^a\mathbf u_a,
 \qquad
 e^a{}_{\mu}=\mathcal N_0^{-1/2}\partial_\mu Y^a,
 \qquad
 e^a=\mathcal N_0^{-1/2}dY^a,
 \label{eq:gradient-coframe}
\end{equation}
with $\det(\partial_\mu Y^a)\neq0$.  The induced metric is
\begin{equation}
 g_{\mu\nu}=\eta_{ab}e^a{}_{\mu}e^b{}_{\nu}
 =\mathcal N_0^{-1}\eta_{ab}\partial_\mu Y^a\partial_\nu Y^b.
 \label{eq:pullback-metric}
\end{equation}
This is an auxiliary restriction obtained by replacing the canonical
covariant jet $D_\mu\Theta$ by the exact gradient $\partial_\mu\Theta$.

\begin{theorem}[Exact-gradient flatness no-go]
\label{thm:gradient-flatness}
Every nondegenerate metric of the form \eqref{eq:pullback-metric} is locally
flat:
\begin{equation}
 R^\rho{}_{\sigma\mu\nu}[g]=0,
 \qquad
 R^{ab}[\mathring\omega(e)]=0.
\end{equation}
Consequently the gradient-composite restriction cannot represent a generic
curved GR geometry.
\end{theorem}

\begin{proof}
Nondegeneracy makes $Y=(Y^0,Y^1,Y^2,Y^3)$ a local coordinate map.  In the
coordinates $Y^a$ the metric is simply $\mathcal N_0^{-1}\eta_{ab}$, so its
Riemann tensor vanishes.  Equivalently, the coframe is exact,
$d e^a=\mathcal N_0^{-1/2}d^2Y^a=0$.  The zero spin connection is
metric-compatible and satisfies the torsion-free Cartan equation
$d e^a+\mathring\omega^a{}_b\wedge e^b=0$; uniqueness of the Levi--Civita spin
connection therefore gives $\mathring\omega^a{}_b=0$ in this coframe and
$R^{ab}=0$.
\end{proof}

\begin{corollary}[Hilbert--Palatini degeneracy]
For every admissible configuration of the gradient-composite restriction,
\begin{equation}
 \epsilon_{abcd}\,e^a\wedge e^b\wedge
 R^{cd}(\mathring\omega(e))=0.
\end{equation}
Thus the Einstein/Hilbert--Palatini term carries no local dynamics on this
restriction, not merely at the affine background.
\end{corollary}

\section{The volume terms and affine stationarity}
The central metric lock gives
\begin{equation}
 \langle\partial_\mu\Theta,\partial_\nu\Theta\rangle_\sharp
 =\mathcal N_0 g_{\mu\nu},
\end{equation}
so in four dimensions
\begin{equation}
 \frac12\sqrt{-g}\,g^{\mu\nu}
 \langle\partial_\mu\Theta,\partial_\nu\Theta\rangle_\sharp
 =2\mathcal N_0\sqrt{-g}.
\end{equation}
On an orientation-preserving patch,
\begin{equation}
 \sqrt{-g}
 =\det(e^a{}_{\mu})
 =\mathcal N_0^{-2}\det(\partial_\mu Y^a).
\end{equation}
Therefore the complete restricted action with $V\equiv0$ is a coefficient
multiple of the Jacobian integral
\begin{equation}
 \int d^4x\,\det(\partial_\mu Y^a),
\end{equation}
which is a null Lagrangian: its first variation is a boundary term by the
Piola identity.  In particular the affine map
\begin{equation}
 \Theta_{\rm aff}=\Theta_0+
 \sqrt{\mathcal N_0}\,\bar E_\mu x^\mu
\end{equation}
is stationary under compactly supported Lorentz-real variations for all
$\Lambda$, $\kappa$, and $\mathcal N_0$.

\begin{corollary}[Correct status of affine admissibility]
The earlier statement called \texttt{GAP-10T-COMPOSITE-FLAT} is true only as
an auxiliary corollary of Theorem~\ref{thm:gradient-flatness}.  It does not
select a physical branch and does not evade the curved-space integrability
problem.
\end{corollary}

\section{Relation to canonical UBT}
Canonical UBT defines
\begin{equation}
 E_\mu=\mathcal N_0^{-1/2}D_\mu\Theta,
\end{equation}
where the connection is reconstructed from the tetrad and torsion and hence
enters the defining jet self-consistently.  Replacing $D_\mu$ by $\partial_\mu$
removes precisely the non-holonomic structure capable of carrying curvature.
Accordingly:
\begin{itemize}
 \item \texttt{GAP-10T-GRADIENT-FLATNESS} is \textbf{CLOSED AS NO-GO [L1]};
 \item gradient-composite affine stationarity is \textbf{CLOSED AS AN
       AUXILIARY COROLLARY [L1]};
 \item the actual self-consistent $D$-composite variation, curved dynamics,
       torsion selection, and coefficient origin remain open under
       \texttt{GAP-10T-DYN} and \texttt{GAP-10D}.
\end{itemize}
