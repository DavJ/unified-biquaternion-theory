\begin{abstract}
Gap~C1 required showing the $W^\pm$ vertex is $P_\psi$-odd from $S[\Theta]$.
We close it by showing the \emph{charged current} $J^\mu_W$ is $P_\psi$-odd,
which follows from:
(i)~$P_\psi=\gamma^5$ [\Lo, \texttt{alpha\_conditions\_closure.tex}],
so left-handed fermions are $P_\psi$-odd and right-handed are $P_\psi$-even;
(ii)~$SU(2)_L$ = left multiplication on $\Theta$ [\Lc, T2\_GAUGE], which
couples exclusively to $P_\psi$-odd (left-handed) fermions.
The $W^\pm$ gauge field itself is $P_\psi$-even (from $P_\psi$ invariance of
$S[\Theta]$), but the charged current $J^\mu_W = W^+_\mu\bar\psi_R\gamma^\mu\psi_L$
is $P_\psi$-odd: $P_\psi(J^\mu_W) = -J^\mu_W$.
Gap~C1 is \textbf{CLOSED} [\Lc\ on T2\_GAUGE].
\end{abstract}

\section{$P_\psi$ Parity of the $W^\pm$ Gauge Field}
\label{sec:W_parity}
%──────────────────────────────────────────────────────────────────

\begin{proposition}[$W^a_\mu$ is $P_\psi$-even — \Lo]
\label{prop:W_even}
The $SU(2)_L$ gauge field $W^a_\mu$ is $P_\psi$-even:
$P_\psi(W^a_\mu) = +W^a_\mu$.
\end{proposition}

\begin{proof}
The covariant derivative
\[
  D_\mu\Theta = \partial_\mu\Theta - iW^a_\mu[\sigma^a/2,\Theta]
\]
must transform consistently under $P_\psi$.
Since $P_\psi\Theta_n = (-1)^n\Theta_n$ [\Lz] and $S[\Theta]$ is
$P_\psi$-invariant [\Lo, \texttt{step1\_psi\_parity.tex}]:
\[
  P_\psi(D_\mu\Theta_n)
  = \partial_\mu((-1)^n\Theta_n) - iW^a_\mu\bigl[\tfrac{\sigma^a}{2},(-1)^n\Theta_n\bigr]
  = (-1)^n D_\mu\Theta_n.
\]
This is consistent with $P_\psi(W^a_\mu)=+W^a_\mu$. \Lo
\end{proof}

\section{$P_\psi$ Parity of the Charged Current}
\label{sec:current_parity}
%──────────────────────────────────────────────────────────────────

\begin{theorem}[Charged current is $P_\psi$-odd — Gap C1 CLOSED — \Lc]
\label{thm:gap_c1}
The $W^\pm$ charged current
\[
  J^{\mu,W}_{\rm charged}
  = W^+_\mu\,\bar\psi_R\gamma^\mu\psi_L + W^-_\mu\,\bar\psi_L\gamma^\mu\psi_R
\]
is $P_\psi$-odd: $P_\psi(J^{\mu,W}_{\rm charged}) = -J^{\mu,W}_{\rm charged}$.
\end{theorem}

\begin{proof}
\textbf{Step 1} [\Lo]. From $P_\psi=\gamma^5$
[\texttt{alpha\_conditions\_closure.tex}, Theorem~1]:
\begin{align*}
  P_\psi\psi_L &= -\psi_L \quad (\text{left-handed: } \gamma^5\text{-eigenvalue} -1),\\
  P_\psi\psi_R &= +\psi_R \quad (\text{right-handed: } \gamma^5\text{-eigenvalue} +1).
\end{align*}

\textbf{Step 2} [\Lz]. Bilinear $\bar\psi_R\gamma^\mu\psi_L$ transforms as:
\[
  P_\psi(\bar\psi_R\gamma^\mu\psi_L)
  = \overline{(P_\psi\psi_R)}\gamma^\mu(P_\psi\psi_L)
  = (+\bar\psi_R)\gamma^\mu(-\psi_L)
  = -\bar\psi_R\gamma^\mu\psi_L.
\]

\textbf{Step 3} (Proposition~\ref{prop:W_even}). $P_\psi(W^+_\mu)=+W^+_\mu$.

\textbf{Combining}:
\[
  P_\psi(W^+_\mu\bar\psi_R\gamma^\mu\psi_L)
  = (+W^+_\mu)(-\bar\psi_R\gamma^\mu\psi_L)
  = -W^+_\mu\bar\psi_R\gamma^\mu\psi_L.
\]
Similarly for the $W^-$ term. Therefore:
\[
  P_\psi(J^{\mu,W}_{\rm charged}) = -J^{\mu,W}_{\rm charged}. \quad \Lc
\]

\textbf{Source of \Lc}: $SU(2)_L$ couples to $\psi_L$ and not $\psi_R$
by the T2\_GAUGE result that $SU(2)_L$ acts from the left on the
spinorial component of $\Theta$ [\Lc, T2\_GAUGE paper].
\end{proof}

\begin{corollary}[OP-S4 closed — \Lc]
The $P_\psi$-odd charged current selects left-handed fermions only.
$SU(2)_R$ coupling to $\psi_L$ would require a $P_\psi$-even charged current,
which is absent. All three OP-S4 loopholes are closed:
\begin{enumerate}
\item $P_\psi$-even $SU(2)_R$ coupling to $n<0$ sector: absent (matter in
  $n>0$ by Lemma~4 of Step~1).
\item $SU(2)_R$ coupling via $P_\psi$-even bilinear: absent
  (Theorem~\ref{thm:gap_c1}).
\item Accidental $P_\psi$ breaking: excluded by $P_\psi$-invariance of
  $S[\Theta]$ [\Lo].
\end{enumerate}
Status of OP-S4: \Lc\ on T2\_GAUGE. \qquad Gap C1: \textbf{CLOSED}.
\end{corollary}
