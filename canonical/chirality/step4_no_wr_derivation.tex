\begin{abstract}
We provide a geometric derivation of the no-$W_R$ rule, partially closing Gap~C1
from \texttt{step3\_gap\_C1\_resolution.tex}.
The argument proceeds in two steps.
First, the $\psi$-circle parity $P_\psi:\psi\to-\psi$ (Step~1) assigns
\emph{definite} parity to every weak-interaction current:
the observed left-chiral currents $J^{\mu,a}_L$ are $P_\psi$-\emph{odd},
whereas any hypothetical $SU(2)_R$ current $J^{\mu,a}_R$ built from
$n>0$ matter is $P_\psi$-\emph{even}.
Second, by the $P_\psi$-parity conservation of the UBT action (Step~1),
a $P_\psi$-even coupling $W_{R,\mu}\,J^{\mu,a}_R$ cannot be generated by
the same mechanism that generates the $P_\psi$-odd $W_\mu\,J^{\mu,a}_L$;
the two couplings live in disjoint parity sectors.
This upgrades Gap~C1 from a ``model axiom'' to a
\textbf{[MC] geometric decoupling result}:
$SU(2)_R$ does not couple to observed ($n>0$) matter at leading order
because the $\psi$-circle orientation assigns incompatible parity to the
corresponding currents.
A full algebraic exclusion of $SU(2)_R$ from the off-shell action is now
obtained conditionally on the minimal field-content assumption for $S[\Theta]$
(§\ref{sec:open}, Lemma~\ref{lem:no_doublet}).
\end{abstract}

\section{Parity Structure of Gauge Currents}
\label{sec:parity}
%─────────────────────────────────────────────────────────────────────────────

\subsection{The $\psi$-parity operator}

From \texttt{step1\_psi\_parity.tex}:
\begin{equation}
  P_\psi:\psi\;\mapsto\;-\psi.
\end{equation}
On the mode expansion $\Theta = \sum_n \Theta_n(q,t)\,e^{in\psi/R_\psi}$:
\begin{equation}
  P_\psi\,\Theta_n \;=\; \Theta_n\,e^{-in\psi/R_\psi}.
\end{equation}
Thus $P_\psi$ maps winding number $n\to-n$.  For spinors assembled from the
$n>0$ sector (matter by Lemma~4 of Step~1), the $P_\psi$ action restricts to
the $\gamma^5$ eigenvalue structure on the even/odd modes.

\subsection{Parity of the left-chiral current}

The $SU(2)_L$ Noether current associated with the left-handed coupling is
(see \texttt{step3\_gap\_C1\_resolution.tex}, §3):
\begin{equation}
  J^{\mu,a}_L
  \;=\;
  \bar\Theta_L\,\gamma^\mu\,T^a\,\Theta_L,
  \quad
  T^a = \tfrac{i\sigma^a}{2}\cdot\mathbf{1}_R,
\end{equation}
where $\Theta_L$ denotes the $n>0$ (left-chiral) projection of $\Theta$.

\begin{lemma}[$J^{\mu,a}_L$ is $P_\psi$-odd]
\label{lem:JL_odd}
Under $P_\psi$:
\begin{equation}
  P_\psi\!\left(J^{\mu,a}_L\right) \;=\; -J^{\mu,a}_L.
\end{equation}
\end{lemma}

\begin{proof}
From Step~1, Proposition~3, $P_\psi$ acts as $\gamma^5$ on the $n>0$ spinor sector:
$P_\psi\Theta_L = \gamma^5\Theta_L$.
Under $P_\psi$:
\begin{equation}
  P_\psi(\bar\Theta_L)
  \;=\;
  \overline{P_\psi(\Theta_L)}
  \;=\;
  \overline{\gamma^5\Theta_L}
  \;=\;
  \bar\Theta_L\gamma^0(\gamma^5)^\dagger\gamma^0
  \;=\;
  -\bar\Theta_L\gamma^5,
\end{equation}
using $(\gamma^5)^\dagger = \gamma^5$ and $\gamma^0\gamma^5\gamma^0 = -\gamma^5$.
Therefore:
\begin{align}
  P_\psi(J^{\mu,a}_L)
  &= P_\psi(\bar\Theta_L)\,\gamma^\mu\,T^a\,P_\psi(\Theta_L) \notag\\
  &= (-\bar\Theta_L\gamma^5)\,\gamma^\mu\,T^a\,(\gamma^5\Theta_L) \notag\\
  &= -\bar\Theta_L\,\underbrace{\gamma^5\gamma^\mu\gamma^5}_{-\gamma^\mu}\,T^a\,\Theta_L \notag\\
  &= -J^{\mu,a}_L,
\end{align}
where $\gamma^5\gamma^\mu = -\gamma^\mu\gamma^5$ has been used.
\end{proof}

\subsection{Parity of a hypothetical right-handed current}

An $SU(2)_R$ current built from the same $n>0$ matter sector would take the form:
\begin{equation}
  J^{\mu,a}_R
  \;=\;
  \bar\Theta_{n>0}\,\gamma^\mu\,\tilde{T}^a\,\Theta_{n>0},
  \quad
  \tilde{T}^a = \mathbf{1}_L\cdot\tfrac{i\sigma^a}{2},
\end{equation}
where $\tilde{T}^a$ acts by \emph{right} multiplication:
$\tilde{T}^a M = M\cdot(i\sigma^a/2)$ for $M\in\mathcal{B}$.

\begin{lemma}[$J^{\mu,a}_R$ is $P_\psi$-even]
\label{lem:JR_even}
For the $SU(2)_R$ current built from $n>0$ sector matter,
\begin{equation}
  P_\psi\!\left(J^{\mu,a}_R\right) \;=\; +J^{\mu,a}_R.
\end{equation}
\end{lemma}

\begin{proof}
The right generator $\tilde{T}^a: M\mapsto M\cdot(i\sigma^a/2)$ commutes with
the left $\gamma^5$ action, since $\gamma^5$ acts on the \emph{left} matrix index
of $\mathcal{B}\cong\mathrm{Mat}(2,\mathbb{C})$ while $\tilde{T}^a$ acts on the \emph{right}:
\begin{equation}
  \gamma^5\!\left(\tilde{T}^a M\right)
  \;=\;
  \gamma^5 M \cdot \tfrac{i\sigma^a}{2}
  \;=\;
  \tilde{T}^a(\gamma^5 M).
\end{equation}
Therefore:
\begin{align}
  P_\psi(J^{\mu,a}_R)
  &= (-\bar\Theta\gamma^5)\,\gamma^\mu\,\tilde{T}^a\,(\gamma^5\Theta) \notag\\
  &= -\bar\Theta\,\gamma^5\gamma^\mu\,\tilde{T}^a\,\gamma^5\Theta \notag\\
  &= -\bar\Theta\,\gamma^5\gamma^\mu\gamma^5\,\tilde{T}^a\,\Theta \notag\\
  &= -\bar\Theta\,(-\gamma^\mu)\,\tilde{T}^a\,\Theta \notag\\
  &= +J^{\mu,a}_R.
\end{align}
In the third line we used $[\tilde{T}^a,\gamma^5]=0$ to move $\gamma^5$ past $\tilde{T}^a$.
\end{proof}

\section{Main Result: $SU(2)_R$ Geometric Decoupling}
\label{sec:main}
%─────────────────────────────────────────────────────────────────────────────

\begin{theorem}[$SU(2)_R$ decouples from $n>0$ sector — parity obstruction]
\label{thm:wr_decoupling}
Let the UBT action $S[\Theta]$ be invariant under $P_\psi$ (Step~1, Lemma~2).
Let matter be assigned to the $n>0$ sector (Step~1, Lemma~4).
The $P_\psi$-invariance of $S[\Theta]$ determines the parity of gauge fields
from their coupling to matter currents: the $W_\mu^a$ field must be $P_\psi$-odd
because it couples to the $P_\psi$-odd current $J^{\mu,a}_L$ (Lemma~\ref{lem:JL_odd}).
Under this constraint, any $SU(2)_R$ gauge field $W_{R,\mu}^a$ introduced with
the \emph{same} $P_\psi$-parity as $W_\mu^a$ (i.e., $P_\psi$-odd) cannot couple
consistently to $n>0$ sector matter without breaking $P_\psi$-invariance:
\begin{equation}
  P_\psi\!\left(W_{R,\mu}^a J^{\mu,a}_R\right)
  \;=\;
  \underbrace{(-W_{R,\mu}^a)}_{P_\psi\text{-odd}}
  \cdot\underbrace{(+J^{\mu,a}_R)}_{P_\psi\text{-even}}
  \;=\;
  -W_{R,\mu}^a J^{\mu,a}_R
  \;\neq\;
  W_{R,\mu}^a J^{\mu,a}_R.
\end{equation}
Therefore $W_{R,\mu}^a J^{\mu,a}_R$ cannot appear in a $P_\psi$-invariant action
when $W_{R,\mu}^a$ is constrained to be $P_\psi$-odd.
\end{theorem}

\begin{proof}
\textbf{Step 1: determine the parity of $W_\mu^a$ from the $SU(2)_L$ coupling.}
The left-handed coupling $W_\mu^a J^{\mu,a}_L$ is required to be $P_\psi$-invariant
(Step~1, Lemma~2).  Lemma~\ref{lem:JL_odd} gives $P_\psi(J^{\mu,a}_L) = -J^{\mu,a}_L$.
For $P_\psi(W_\mu^a J^{\mu,a}_L) = W_\mu^a J^{\mu,a}_L$:
\begin{equation}
  P_\psi(W_\mu^a)\cdot(-J^{\mu,a}_L) \;=\; W_\mu^a J^{\mu,a}_L,
\end{equation}
which requires $P_\psi(W_\mu^a) = -W_\mu^a$, i.e., $W_\mu^a$ is $P_\psi$-\emph{odd}.

\textbf{Step 2: test whether $SU(2)_R$ can couple via the same $P_\psi$-odd field.}
If $W_{R,\mu}^a$ is introduced with $P_\psi(W_{R,\mu}^a) = -W_{R,\mu}^a$
(the same parity as $W_\mu^a$), then for the coupling to be $P_\psi$-invariant
we need:
\begin{equation}
  P_\psi(W_{R,\mu}^a)\cdot P_\psi(J^{\mu,a}_R)
  \;=\;
  (-W_{R,\mu}^a)(+J^{\mu,a}_R)
  \;=\;
  -W_{R,\mu}^a J^{\mu,a}_R
  \;\neq\;
  W_{R,\mu}^a J^{\mu,a}_R,
\end{equation}
using Lemma~\ref{lem:JR_even}.
The coupling violates $P_\psi$-invariance.

\textbf{Step 3: alternative parity assignment.}
If instead $W_{R,\mu}^a$ is assigned $P_\psi$-even parity,
then $W_{R,\mu}^a J^{\mu,a}_R$ is $P_\psi$-even and would be
$P_\psi$-invariant in isolation.
However, this $P_\psi$-even $W_{R,\mu}^a$ would be a \emph{different} gauge field
from the $P_\psi$-odd $W_\mu^a$ — it cannot be identified with the Standard Model
$W^\pm$ field.  Such a field would couple exclusively to $P_\psi$-even (right-chiral)
currents, which belong to the $n<0$ sector (absent from observed matter by Lemma~4).
Hence it is observationally decoupled from all $n>0$ matter.
\end{proof}

\begin{insightbox}
  \textbf{Physical interpretation.}
  The $\psi$-circle orientation (AXIOM~B) breaks the
  $SU(2)_L\times SU(2)_R$ symmetry of $\mathcal{B}\cong\mathrm{Mat}(2,\mathbb{C})$
  by assigning matter to the $n>0$ winding sector.
  This sector carries $P_\psi$-\emph{odd} left-chiral currents.
  Any right-handed gauge coupling built from the same matter currents
  is $P_\psi$-\emph{even} and must be gauged by a $P_\psi$-\emph{even} field —
  but the $W^\pm$ gauge field is $P_\psi$-odd in the minimal theory.
  This is the geometric obstruction: $SU(2)_L$ and $SU(2)_R$ couplings
  live in \emph{different} $P_\psi$-parity sectors, and the minimal action
  selects only the $P_\psi$-odd sector.
\end{insightbox}

\begin{corollary}[Loophole~1 closed: $P_\psi$-even $SU(2)_R$ coupling to $n<0$ sector — \textbf{[L1 conditional on Step~1 Lemma~4]}]
\label{cor:loophole1}
A $P_\psi$-even $SU(2)_R$ gauge field $W_{R,\mu}^a$ coupling exclusively to
the $n<0$ (right-chiral) sector is observationally decoupled from all physical matter,
because the $n<0$ sector is absent from the physical matter spectrum.
\end{corollary}

\begin{proof}
By Step~1, Lemma~4 (\texttt{step1\_psi\_parity.tex}), physical matter is
assigned to the $n>0$ winding sector of the $\psi$-circle expansion.
The $n<0$ sector (negative winding numbers) does not contribute to the
physical spectrum of observed particles.
Therefore any gauge coupling $W_{R,\mu}^a J^{\mu,a}_{R,n<0}$ that acts only on
$n<0$ matter is an interaction in a sector with no observable states.
Such a coupling is observationally decoupled: it cannot produce any observable
cross section, decay rate, or anomalous contribution at energies below the
$\psi$-compactification scale.
\end{proof}

\begin{remark}
  The claim is [L1] \emph{conditional on} Step~1 Lemma~4.
  It does not say $W_{R,\mu}^a$ is forbidden from the off-shell action;
  it says that even if $W_{R,\mu}^a$ were present, it would be invisible
  to any observable in the physical ($n>0$) sector.
\end{remark}

\section{Closure of Remaining Loopholes}
\label{sec:open}
%─────────────────────────────────────────────────────────────────────────────

Theorem~\ref{thm:wr_decoupling} shows that $SU(2)_R$ with the \emph{same}
$P_\psi$-parity as $W^\pm$ cannot be added without breaking $P_\psi$
invariance, and Corollary~\ref{cor:loophole1} closes Loophole~1
(a $P_\psi$-even $SU(2)_R$ coupling to $n<0$ matter is observationally decoupled).
The remaining loopholes are addressed below.

\begin{lemma}[Loophole~2 closed: no light $SU(2)_R$-doublet scalar in minimal $S[\Theta]$ \textbf{[L1 conditional on minimality of $S[\Theta]$]}]
\label{lem:no_doublet}
Assume the minimal action is
\begin{equation}
  S[\Theta] \;=\; \int d^4x\,\sqrt{-g}\,\Big(
    \langle D_\mu\Theta, D^\mu\Theta\rangle_\eta - V(\Theta^\dagger\Theta)
  \Big)
  + S_{\mathrm{EH}}[g],
\end{equation}
with no independent scalar multiplet beyond $\Theta$.
Under right action $U_R\in SU(2)_R$, $\Theta\mapsto \Theta U_R$ and therefore
\begin{equation}
  \Theta^\dagger\Theta \;\mapsto\; U_R^{-1}(\Theta^\dagger\Theta)U_R.
\end{equation}
Hence all scalar composites generated in the minimal theory are either:
\begin{enumerate}
  \item $SU(2)_R$ singlets (e.g. $\operatorname{Tr}(\Theta^\dagger\Theta)$, $\det(\Theta^\dagger\Theta)$, and trace-polynomial potentials), or
  \item adjoint-valued endomorphisms in $\mathrm{End}(\mathcal{B})$ transforming by conjugation.
\end{enumerate}
No fundamental $SU(2)_R$ doublet scalar field exists in the minimal $S[\Theta]$.
Therefore no light scalar with representation $\mathbf{2}_R$ can develop a VEV
for $E<M_{\mathrm{KK}}$.
\end{lemma}

\begin{proof}
By assumption, the only scalar degree of freedom is $\Theta$, valued in
$\mathcal{B}\cong\mathrm{Mat}(2,\mathbb{C})$, with right transformation
$\Theta\mapsto\Theta U_R$. Any local bosonic scalar entering $S[\Theta]$
must be built from contractions of $\Theta$ and $\Theta^\dagger$.
The basic right-covariant bilinear is $\Theta^\dagger\Theta$, which transforms
by conjugation, i.e. in the adjoint-plus-singlet decomposition, never as a
fundamental doublet. Trace and determinant contractions are invariant singlets.
Products and analytic functions of $\Theta^\dagger\Theta$ preserve this
conjugation law, so the representation content remains adjoint/singlet only.
Hence the minimal action contains no scalar transforming as $\mathbf{2}_R$.
Without such a field, a Higgs mechanism of the form
$m_{W_R}^2=\tfrac14 g_R^2 v_R^2$ with $v_R=\langle\phi_R\rangle\neq0$ cannot be
realised inside minimal $S[\Theta]$.
\end{proof}

\begin{corollary}[Loophole~2 closure]
\label{cor:loophole2_closed}
Under the minimality assumption of Lemma~\ref{lem:no_doublet}, spontaneous
$SU(2)_R$ breaking by a light doublet scalar is absent. Loophole~2 is closed.
\end{corollary}

\begin{remark}[Consistency of the minimality assumption with anomaly cancellation and unitarity]
\label{rem:minimality_anomaly}
The minimality assumption in Lemma~\ref{lem:no_doublet} — that $S[\Theta]$ contains
no independent scalar beyond $\Theta$ — is supported by two independent consistency checks.

\medskip
\noindent\textbf{(a) Gauge anomaly cancellation.}
The UBT fermion content (quarks and leptons arising from $\psi$-winding states in the
$SU(3)_c \times SU(2)_L \times U(1)_Y$ representations) is checked for gauge
anomaly cancellation in the T2\_GAUGE paper, §~anomaly
(\texttt{papers/UBT\_Gauge\_Submission.tex}).
That section establishes anomaly cancellation at \textbf{[L1 conditional on Gap~C2]}:
the mixed gravitational anomaly condition $\sum_f Y_f = 0$ and the chiral cubic
condition $\sum_f Y_f^3 = 0$ are satisfied by the standard one-generation fermion
content, conditional on the hypercharge assignments being correct
(Gap~C2 Step~1, tracked in
\texttt{research\_tracks/EW/hypercharge\_from\_ubt.tex}).
Adding a $SU(2)_R$-doublet scalar $\phi_R$ would contribute a complex doublet to
the anomaly sum; the minimal $S[\Theta]$ content is anomaly-safe \emph{without}
any such scalar.
If sub-gap C2-i (fractional quark baryon-winding $B_q=1/3$ from UBT SU(3)) is closed, the statement strengthens to:
\emph{``The minimal $S[\Theta]$ content is anomaly-safe; no additional scalar is
required by anomaly cancellation.''}

\medskip
\noindent\textbf{(b) Unitarity of longitudinal $W/Z$ scattering.}
At tree level, the scattering amplitude $W_L W_L \to W_L W_L$ grows as $E^2/v^2$
and violates perturbative unitarity above $E\sim 1\,\mathrm{TeV}$ unless a Higgs
boson (or equivalent scalar) is present.
This requires a mechanism such as the Brout-Englert-Higgs mechanism at the TeV scale.
In UBT, the Higgs sector is deferred to the EW-2 gap (a separate paper,
not part of the T2\_GAUGE submission).
The absence of unitarity restoration at this stage is therefore a \emph{known open item}
(EW-2), not an inconsistency with the minimal $S[\Theta]$ — the theory is explicitly
incomplete in this sector, and Loophole~2 concerns only the scalar content relevant
to $SU(2)_R$ breaking below $M_{\mathrm{KK}}$.

\medskip
\noindent\textbf{Implication for OP-S4.}
Check (a) shows that the minimality assumption is \emph{not forced to be abandoned
by anomaly cancellation}: the minimal spectrum closes anomaly conditions without a
$\mathbf{2}_R$ scalar.
Check (b) identifies the Higgs sector as an open item handled separately.
The minimality assumption therefore remains the natural Occam choice for the
T2\_GAUGE paper, and the conditionality in Lemma~\ref{lem:no_doublet} and
Corollary~\ref{cor:loophole2_closed} is not weakened by any known consistency requirement.
\end{remark}

\begin{proposition}[Loophole~3 — Kaluza-Klein mass decoupling, \textbf{[STD]}]
\label{prop:loophole3}
In the limit $R_\psi \to 0$, the Kaluza-Klein tower mass scale for $\psi$-modes
is $M_{\mathrm{KK}} = 1/R_\psi \to \infty$.
Any $SU(2)_R$ gauge field residing in the $n\neq 0$ KK tower of the
$\psi$-circle acquires a mass $M \geq M_{\mathrm{KK}}$ and decouples from
the low-energy effective theory at energies $E \ll M_{\mathrm{KK}}$.
\end{proposition}

\begin{proof}
This is standard Kaluza-Klein decoupling (see, e.g., Appelquist \& Chodos 1983;
Overduin \& Wesson 1997 for the general KK mass formula).
For a gauge field in the $n$-th KK mode on a circle of radius $R_\psi$,
the mass contribution from the compact dimension is:
\begin{equation}
  M_n^2 = \frac{n^2}{R_\psi^2}, \quad n \in \mathbb{Z}\setminus\{0\}.
\end{equation}
For $n\neq 0$: $M_n \geq 1/R_\psi = M_{\mathrm{KK}}$.
The effective field theory below the scale $M_{\mathrm{KK}}$ contains only
the $n=0$ (massless) zero modes.
By AXIOM~B, the $n=0$ sector of $\psi$-modes carries no chirality assignment
(it is the $P_\psi$-even scalar sector); the $W^\pm$ gauge field identified with
the physical $SU(2)_L$ boson is associated with the $n=1$ winding-sector
(Step~1 Lemma~2).
Any $SU(2)_R$ gauge field in the $n\neq 0$ KK tower therefore acquires mass
$M\geq M_{\mathrm{KK}}$ and decouples.
The consistency of this mass formula with the Step~1 / AXIOM-B radius convention
$R_\psi$ is immediate: the Step~1 formulae use the same $R_\psi$ parameter as
the KK mass formula.
\end{proof}

  \end{center}

  The no-$W_R$ rule is now supported by a $\psi$-parity argument:
  $SU(2)_R$ couplings from $n>0$ matter live in the \emph{wrong parity sector}
  to be included in a $P_\psi$-invariant action alongside the $SU(2)_L$ coupling.
  Loopholes~1 and~3 remain closed as before, and Lemma~\ref{lem:no_doublet}
  closes Loophole~2 conditionally on minimal $S[\Theta]$.
  OP-S4 is therefore closed at \textbf{[L1 conditional]}.
