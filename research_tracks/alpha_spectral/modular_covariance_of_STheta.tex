\begin{abstract}
We test whether the current UBT action $S[\Theta]$ is modular-covariant on the
complex-time torus, with explicit checks of: (i) the modulus definition
$\tau=t+i\psi$, (ii) the $\SL(2,\ZZ)$ action on $\tau$, (iii) induced
transformation of Fourier modes, (iv) covariance of $\nabla^\dagger\nabla$,
(v) transformation of the measure $\dd^4x\,\dd\psi$, and (vi) existence of
equal-action Hecke saddles. No $\alpha$ input, no fixed-prime input, and no
$\eta(i)$ fitting are used.
\end{abstract}

\section{Setup}

\begin{definition}[Complex-time modulus]
We take
\begin{equation}
  \tau = \frac{t+i\psi}{\beta} \in \HH,\qquad \Im\tau>0,
\end{equation}
with compact imaginary-time direction $\psi\sim\psi+2\pi R_\psi$.
\end{definition}

\begin{definition}[Candidate action]
The currently used action-level structure in this track is
\begin{equation}
  S[\Theta]
  = \int_{\mathcal M_4\times S^1_\psi}\dd^4x\,\dd\psi\;
    \mathcal L\!\left(\Theta,\nabla^\dagger\nabla\Theta\right),
  \label{eq:action}
\end{equation}
with kinetic core governed by $\nabla^\dagger\nabla$.
\end{definition}

\section{$\SL(2,\ZZ)$ action on the modulus}

For $\gamma=\begin{psmallmatrix}a&b\\c&d\end{psmallmatrix}\in\SL(2,\ZZ)$:
\begin{equation}
  \tau\mapsto\gamma\cdot\tau=\frac{a\tau+b}{c\tau+d},
  \qquad
  \Im(\gamma\cdot\tau)=\frac{\Im\tau}{|c\tau+d|^2}.
  \label{eq:im-transform}
\end{equation}

\begin{proposition}[Jacobian seen by the compact direction]
If $\psi$ is tied to $\Im\tau$ at fixed real-time scale, then under
\eqref{eq:im-transform} one gets
\begin{equation}
  \dd\psi \mapsto \dd\psi' = |c\tau+d|^{-2}\dd\psi.
  \label{eq:dpsi-transform}
\end{equation}
\end{proposition}

\section{Induced transformation of $\Theta$ modes}

Write the torus-mode expansion in cycle coordinates $(u,v)\in\mathbb T^2$:
\begin{equation}
  \Theta(x;u,v)=\sum_{(m,n)\in\ZZ^2}\Theta_{m,n}(x)\,
  e^{2\pi i(mu+nv)}.
\end{equation}
Under cycle change by $\gamma$, integer labels transform contragrediently:
\begin{equation}
  (m,n)\mapsto(m,n)\gamma^{-1},
  \qquad
  \Theta'_{m,n}(x)=\Theta_{(m,n)\gamma^{-1}}(x).
  \label{eq:mode-relabel}
\end{equation}
So modular action gives a relabelling/permutation of lattice modes unless extra
field weight is postulated.

\section{Covariance test for $\nabla^\dagger\nabla$}

\begin{proposition}[What would be required]
For modular covariance of the kinetic part of \eqref{eq:action}, one needs
\begin{equation}
  (\nabla^\dagger\nabla)_\tau
  \;\mapsto\;
  |c\tau+d|^{w}\,U_\gamma^{-1}(\nabla^\dagger\nabla)_\tau U_\gamma
\end{equation}
for some definite weight $w$ and induced mode map $U_\gamma$ compatible with
\eqref{eq:mode-relabel}.
\end{proposition}

\begin{gapbox}
\textbf{Observed status in repository derivations.}
No explicit derivation is present that proves a definite modular weight for
$\nabla^\dagger\nabla$ in the current UBT action framework. Existing analyses
already flag this as the critical obstruction (O1).
\end{gapbox}

\section{Measure test}

Using \eqref{eq:dpsi-transform},
\begin{equation}
  \dd^4x\,\dd\psi \mapsto |c\tau+d|^{-2}\dd^4x\,\dd\psi.
\end{equation}
Therefore strict invariance of the full integrand in \eqref{eq:action}
requires compensating modular weight from the kinetic/potential density and
the functional measure. That cancellation has not been derived.

\section{Hecke equal-action saddle consequence}

Let $\bar\Theta_\gamma$ denote saddle candidates generated by modular images.
Equal-action Hecke saddles require
\begin{equation}
  S[\bar\Theta_\gamma]=S[\bar\Theta],\quad\forall\gamma.
  \label{eq:equal-action}
\end{equation}
Without proved modular covariance of both operator and measure, \eqref{eq:equal-action}
cannot be established from \eqref{eq:action}.

\begin{proposition}[Current-state implication]
At the current derivation level, equal-action Hecke saddle degeneracy is not
derivable from $S[\Theta]$.
\end{proposition}
