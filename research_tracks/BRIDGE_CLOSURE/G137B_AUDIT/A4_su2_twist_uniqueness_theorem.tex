\begin{abstract}
The Scherk--Schwarz boundary condition
$\Theta(\psi+2\pi R_\psi)=\sigma_3\Theta(\psi)\sigma_3$ is not an
independent axiom of UBT. It is the \emph{unique} $\mathbb{Z}_2$
twist on $S^1_\psi$ consistent with three derived properties of
$\Theta\in\mathrm{Mat}(2,\mathbb{C})$:
(i) Hermitian symmetry $\Theta\leftrightarrow\Theta^\dagger$,
(ii) minimality of the KK spectrum ($\mathbb{Z}_2$ order),
(iii) compatibility with the $SU(3)$ colour projector $\Pi_{\rm colour}$.
Status: \Lo.
\end{abstract}

%──────────────────────────────────────────────────────────────────
\section{Setup}
%──────────────────────────────────────────────────────────────────

A boundary condition on $S^1_\psi$ for $\Theta\in\mathrm{Mat}(2,\mathbb{C})$
takes the form
\[
  \Theta(\psi+2\pi R_\psi) = U\,\Theta(\psi)\,V
\]
for $U,V\in U(2)$. We seek the unique BC satisfying (i)--(iii).

%──────────────────────────────────────────────────────────────────
\section{Three Constraints}
%──────────────────────────────────────────────────────────────────

\begin{lemma}[Hermitian symmetry $\Rightarrow$ $V=U^\dagger$ — \Lz]
$\Theta\in\mathrm{Mat}(2,\mathbb C)$ has the natural Hermitian conjugate
$\Theta^\dagger$. The BC must be compatible with
$\Theta\leftrightarrow\Theta^\dagger$:
\[
  (\Theta^\dagger)(\psi+2\pi) = V^\dagger\Theta^\dagger U^\dagger.
\]
For this to equal $U\Theta^\dagger V$ (the conjugate BC), we need
$V^\dagger = U$, i.e.\ $V=U^\dagger$.
Hence the BC reduces to $\Theta\to U\Theta U^\dagger$ with $U\in SU(2)$.
\end{lemma}

\begin{lemma}[$\mathbb{Z}_2$ minimality — \Lz]
The minimal non-trivial twist has order $2$ ($\mathbb{Z}_2$): applying
the BC twice returns $\Theta$. Elements of $SU(2)$ of order $2$ are
exactly $\{U: U^2=\pm\mathbf{1}\} = \{\pm i\sigma_1, \pm i\sigma_2,
\pm i\sigma_3\}/\{\pm\mathbf{1}\}$, i.e.\ conjugates of $i\sigma_k$
for $k=1,2,3$.
\end{lemma}

\begin{lemma}[$SU(3)$ compatibility $\Rightarrow$ $U=\sigma_3$ — \Lz]
The $SU(3)$ colour projector $\Pi_{\rm colour}$ acts on the
diagonal entries of $\Theta$ (the colour-singlet and colour-octet
decomposition) [\Lz, \texttt{su3\_from\_involutions.tex}].
$\Pi_{\rm colour}$ is diagonal in the standard basis.

Under $\Theta\to U\Theta U^\dagger$:
\begin{itemize}
\item $U=\sigma_3=\mathrm{diag}(+1,-1)$: diagonal entries $\to$
  diagonal, off-diagonal $\to$ off-diagonal with sign flip.
  $\Pi_{\rm colour}$ is preserved. \checkmark
\item $U=\sigma_1$ or $\sigma_2$: mixes diagonal and off-diagonal
  entries, breaking the diagonal/off-diagonal split.
  $\Pi_{\rm colour}$ is not preserved. $\times$
\end{itemize}
Hence $\sigma_3$ is the unique $\mathbb{Z}_2$ element consistent with
$\Pi_{\rm colour}$.
\end{lemma}

%──────────────────────────────────────────────────────────────────
\section{Main Result}
%──────────────────────────────────────────────────────────────────

\begin{theorem}[SS BC uniqueness — \Lo]
\label{thm:SSBC}
The unique boundary condition on $S^1_\psi$ for
$\Theta\in\mathrm{Mat}(2,\mathbb C)$ satisfying Hermitian symmetry,
$\mathbb{Z}_2$ minimality, and $SU(3)$ compatibility is:
\[
  \Theta(\psi+2\pi R_\psi) = \sigma_3\,\Theta(\psi)\,\sigma_3. \quad \Lo
\]
\end{theorem}

\begin{proof}
Hermitian symmetry fixes $V=U^\dagger$ [\Lz].
$\mathbb{Z}_2$ minimality restricts to $U\in\{\pm\sigma_k\}$ [\Lz].
$SU(3)$ compatibility selects $U=\sigma_3$ uniquely [\Lz].
The overall sign is fixed by $\sigma_3^2=\mathbf{1}$. \Lo
\end{proof}
