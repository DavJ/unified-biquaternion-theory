\section{EW Symmetry Breaking Sign}
\label{sec:ssb}
%──────────────────────────────────────────────────────────────────

\subsection{Setup}

The minimal EW doublet component $\Phi\in\mathbb{C}^2$ of $\Theta$ has
effective quadratic potential:
\[
  V_{\rm EW}(\Phi)
  = V_0 + m^2_{\rm eff}\Phi^\dagger\Phi
  + \lambda_{\rm EW}(\Phi^\dagger\Phi)^2 + \cdots
\]
EW SSB requires $m^2_{\rm eff}<0$, equivalently $\mu^2_{\rm EW}>0$ in the
convention $m^2_{\rm eff}=-\mu^2_{\rm EW}$.

We write $m^2_{\rm eff} = m^2_{0,\rm EW} - \Gamma_\psi M_\psi^2$ where
$M_\psi = R_\psi^{-1}$ and $\Gamma_\psi$ is the $\psi$-sector spectral
stiffness.

\subsection{Condition H1: Modular criticality $m^2_{0,\rm EW}=0$}

\begin{proposition}[Modular criticality --- \Lc]
\label{prop:H1}
Under T-duality invariance of $S[\Theta]$ (i.e.\ $S[\Theta,R_\psi]
= S[\Theta,1/R_\psi]$), any UV-generated quadratic mass term
$m^2_0 \propto R_\psi^{-2}$ is not T-duality invariant and must vanish
at the self-dual point:
\[
  m^2_{0,\rm EW} = 0. \quad \Lc
\]
\end{proposition}

\begin{proof}
Under $R_\psi\to 1/R_\psi$: $m^2_0\propto R_\psi^{-2}\to R_\psi^2 m^2_0$.
T-duality invariance requires $m^2_0 = R_\psi^2 m^2_0$ for all $R_\psi$,
which forces $m^2_0=0$ (since $R_\psi^2\neq 1$ away from the fixed point).
At the self-dual point $R_\psi=1$ this is consistent with $m^2_0=0$.
The UBT natural UV cutoff is $\Lambda=R_\psi^{-1}$, so $m^2_0\propto\Lambda^2$
is the only candidate hard mass, which vanishes by T-duality. \Lc
\end{proof}

\subsection{Condition H2: Positive spectral stiffness $\Gamma_\psi>0$}

\begin{proposition}[Positive trace condition --- \Lz]
\label{prop:H2}
The odd-winding coupling to the EW doublet is $K_{\rm EW} = e^2>0$,
where $e$ is the EM coupling derived from $|D_\psi\Theta|^2$.
The odd-winding kinetic operator $D_{\rm odd}$ is positive-definite
on the physical Hilbert space. Therefore:
\[
  \mathrm{Tr}(D_{\rm odd}^{-1}K_{\rm EW})
  = e^2\,\mathrm{Tr}(D_{\rm odd}^{-1}) > 0. \quad \Lz
\]
\end{proposition}

\begin{proof}
From $S_\psi[\Theta]\supset|D_\psi\Theta|^2$ where
$D_\psi=\partial_\psi+ieA_\psi$: expanding to quadratic order in $A_\psi$
gives $K_{\rm EW}=e^2>0$ (since $e\in\mathbb{R}$, $e>0$).
$D_{\rm odd}>0$: the odd-winding KK operator has positive eigenvalues
$(n+\tfrac12)^2/R_\psi^2>0$ for all $n\in\mathbb{Z}$.
Hence $D_{\rm odd}^{-1}>0$ and $\mathrm{Tr}(D_{\rm odd}^{-1}K_{\rm EW})>0$.
\Lz
\end{proof}

\begin{theorem}[EW SSB sign --- \Lc]
\label{thm:ssb}
Under H1 (Proposition~\ref{prop:H1}) and H2 (Proposition~\ref{prop:H2}),
together with the canonical bridge result that odd-winding modes carry
Grassmann/Berezin measure \Lc\ \cite{fermionic_bridge}:
\begin{enumerate}
\item The effective quadratic coefficient is
  $m^2_{\rm eff} = 0 - \Gamma_\psi M_\psi^2 = -e^2\,\mathrm{Tr}(D_{\rm odd}^{-1})
  \cdot M_\psi^2 < 0$.
\item Therefore $\mu^2_{\rm EW} = -m^2_{\rm eff} > 0$: EW symmetry is
  spontaneously broken.
\item The minimum satisfies $\Phi^\dagger\Phi = v^2/2 > 0$.
\end{enumerate}
Status: \Lc\ (conditional on odd-winding EW readout and T-duality invariance).
\end{theorem}

The factor $4\pi$ is fixed by the standard compact-U(1) charge
