\begin{abstract}
We derive $R_\psi = 1$ (self-dual point $\tau_E = i$) via two
independent paths.

\textbf{Path 1 — Casimir polynomial \Lz+\MC.}
If the effective action $S_{\mathrm{eff}}$ is invariant under
$R_\psi\to 1/R_\psi$, combining the S-invariance condition with
stationarity yields the polynomial $R_\psi^6+R_\psi^4+R_\psi^2=3$,
whose unique positive solution is $R_\psi=1$.
$N_{\mathrm{eff}}$ cancels; the result holds for any $N_{\mathrm{eff}}$.
The S-invariance of $S_{\mathrm{eff}}$ is assumed [\MC]; the
polynomial itself is an algebraic identity [\Lz].

\textbf{Path 2 — Partition function \Lo\ (conditional).}
The one-loop partition function of the UBT $\psi$-sector is
$Z[\tau_E]=\eta(\tau_E)^{-12}$, derived from
$\det P_{\mathrm{eff}}=\eta(\tau_E)^2$ [\STD] and $N_{\mathrm{eff}}=12$
[\Lo].  The modular identity
$\eta(-1/\tau_E)=\sqrt{-i\tau_E}\,\eta(\tau_E)$ makes $\tau_E=i$
the \emph{unique} fixed point of the S-transformation on the
imaginary axis $\tau_E=iR_\psi$, $R_\psi>0$ [\Lz].
The two conditions are: (i)~$\det P_{\mathrm{eff}}=\eta^2$
[\STD, cited], (ii) one-loop approximation [explicit technical
assumption, standard in QFT].

\textbf{Implications.}  $\psi$ is compact from the Scherk--Schwarz
boundary condition [\Lo] and from the periodicity of $Z[\tau]$
[\Lz], independently of any geometric axiom.  Scale closure
$M_{\mathrm{GUT}}=1/R_\psi$: [\MC].
$\tau=i$ is algebraically preferred for the alpha track [\Lz],
but \textbf{alpha is NOT DERIVED} and G137-B is NOT CLOSED.
\end{abstract}

of the $\psi$-circle $S^1_\psi$ is not derived from $S[\Theta]$

\begin{enumerate}
\item \textbf{Alpha (Gap G137-B)}: $B_{\mathrm{phenom}}$ involves

\item \textbf{Scale closure (EW-1b)}: $M_{\mathrm{GUT}}=1/R_\psi^*$;

This note provides two independent derivations of $R_\psi=1$ and

\section{Setup: Modular Parameter and S-Transformation}
\label{sec:setup}
%──────────────────────────────────────────────────────────────────

Complex time $\tau=t+i\psi\in\mathbb{C}$.  The Euclidean modular
parameter for the $\psi$-circle with period $2\pi R_\psi$:
\[
  \tau_E = iR_\psi \in \mathbb{H}, \qquad
  R_\psi > 0.
\]
The modular S-transformation:
\[
  S:\;\tau_E \;\longmapsto\; -\frac{1}{\tau_E},
  \qquad\text{equivalently}\quad R_\psi \;\longmapsto\; \frac{1}{R_\psi}.
\]
\begin{definition}[Self-dual point]
$\tau_E = i$ ($R_\psi = 1$) satisfies $S(i) = -1/i = i$.
It is the unique fixed point of $S$ on the imaginary axis
$\{iR_\psi : R_\psi > 0\}$. \Lz
\end{definition}

\subsection{Compactness of $\psi$ from the Scherk--Schwarz condition}

\begin{lemma}[Compactness of $\psi$ — \Lo]
\label{lem:SS}
From \textup{\texttt{su2\_twist\_neff12.tex}}, the Scherk--Schwarz
boundary condition on $S^1_\psi$ is~\cite{su2_twist}:
\[
  \Theta(\psi + 2\pi R_\psi) = \sigma_3\,\Theta(\psi)\,\sigma_3,
  \qquad \sigma_3 = \mathrm{diag}(+1,-1).
\]
This condition assigns $\psi$ a natural period $2\pi R_\psi$:
diagonal components $a,d$ are periodic ($m_n = n/R_\psi$);
off-diagonal components $b,c$ are anti-periodic
($m_n = (n+\tfrac{1}{2})/R_\psi$).

\textbf{Consequence}: the compactness of $\psi$ is a property of
the symmetry of the field $\Theta$, not a geometric axiom. \Lo
\end{lemma}

%──────────────────────────────────────────────────────────────────
\subsection{One-loop partition function}

\begin{lemma}[Functional determinant for a scalar on $S^1$ — \STD]
\label{lem:detP}
For a periodic real scalar field on $S^1$ with modular parameter
$\tau_E = iR_\psi$, the $\zeta$-regularised one-loop functional
determinant is~\cite[§7.2]{polchinski_v1}:
\[
  \det P_{\mathrm{eff}}(\tau_E) = \eta(\tau_E)^2,
  \qquad
  \eta(\tau) = q^{1/24}\prod_{n=1}^{\infty}(1-q^n),
  \quad q = e^{2\pi i\tau}.
\]
Hence the one-loop contribution to the effective action is
$\frac{1}{2}\ln\det P_{\mathrm{eff}} = \ln\eta(\tau_E)$.
\end{lemma}

\begin{proposition}[UBT $\psi$-sector partition function — \Lo\ conditional]
\label{prop:Z}
From Lemma~\ref{lem:detP} [\STD] and $N_{\mathrm{eff}}=12$ [\Lo,
\textup{\cite{su2_twist}}], the total one-loop partition function
of the UBT $\psi$-sector is:
\[
  \boxed{Z[\tau_E] \;=\; \eta(\tau_E)^{-12}.}
\]
\textbf{Derivation}: The $N_{\mathrm{eff}}$ effective modes (three
compact phase directions $\times$ two charged fields $\times$ two
Weyl helicity weights from the half-integer tower, all [\Lo,
\cite{su2_twist}]) each contribute $\ln\eta(\tau_E)$ to $-\ln Z$:
\[
  -\ln Z[\tau_E] = N_{\mathrm{eff}}\,\ln\eta(\tau_E).
\]

\textbf{Explicit conditions for this result:}
\begin{enumerate}
\item $\det P_{\mathrm{eff}} = \eta(\tau_E)^2$ [\STD] — standard
  result, cited.
\item $N_{\mathrm{eff}} = 12$ [\Lo] — proved in \cite{su2_twist}.
\item One-loop approximation: higher loops give $O(g^4)$ corrections
  that shift the value of $Z[\tau_E]$ but not the \emph{location} of
  the fixed point $\tau_E=i$ (which is determined by algebra, not by
  the magnitude of $Z$).
\end{enumerate}
Status: \Lo\ conditional on items 1--3.
\end{proposition}

%──────────────────────────────────────────────────────────────────
\subsection{Effective compactification from periodicity of $Z[\tau]$}

\begin{proposition}[Effective compactification — \Lz]
\label{prop:compact}
From the standard identity~\cite[§9.2]{apostol_modular}:
\[
  \eta(\tau_E + 1) = e^{i\pi/12}\,\eta(\tau_E).
\]
For $Z[\tau_E] = \eta(\tau_E)^{-12}$:
\[
  Z[\tau_E + 1] = e^{-i\pi}\,Z[\tau_E] = -Z[\tau_E].
\]
The physical amplitude $|Z[\tau_E+1]|^2 = |Z[\tau_E]|^2$ is
therefore periodic in $\tau_E \to \tau_E + 1$, inducing the
effective identification $\tau_E \sim \tau_E + 1$ as a symmetry of
observables. \Lz

\textbf{Consequence}: $\psi$ is compact both from the SS condition
(Lemma~\ref{lem:SS}, \Lo) and from the periodicity of $Z[\tau]$
(\Lz), independently and consistently.
\end{proposition}

%──────────────────────────────────────────────────────────────────
\subsection{$R_\psi = 1$ as the unique fixed point}

\begin{proposition}[Modular transformation of $S{[\Theta]}$: stationarity at $\tau_E=i$ — \Lz]
\label{prop:Smod}
From~\cite[§9.2]{apostol_modular}:
$\eta(-1/\tau_E) = \sqrt{-i\tau_E}\,\eta(\tau_E)$.
With $S[\Theta] = -\ln Z[\tau_E] = 12\ln\eta(\tau_E)$:
\[
  S\!\left[-\frac{1}{\tau_E}\right]
  = S[\tau_E] + 6\ln(-i\tau_E).
\]
For $\tau_E = iR_\psi$: $-i\tau_E = R_\psi > 0$, so
$\ln(-i\tau_E) = \ln R_\psi$.  At $R_\psi = 1$:
\[
  \ln(-i \cdot i) = \ln 1 = 0
  \;\;\Longrightarrow\;\;
  \left.\frac{d}{d\epsilon}
  S\!\left[-\frac{1}{\tau_E+\epsilon}\right]\right|_{\tau_E=i,\,\epsilon=0}
  = 0.
\]
The physical action $\mathrm{Re}\,S[\Theta]$ is stationary under the
modular S-flow at $\tau_E = i$. \Lz
\end{proposition}

\begin{theorem}[$R_\psi = 1$ from partition function symmetry — \Lo\ conditional]
\label{thm:fixed}
Under the conditions of Proposition~\ref{prop:Z}:
\begin{enumerate}
\item \textbf{Fixed point}: $\tau_E = i$ satisfies $S(\tau_E)=\tau_E$,
  hence the factor $(-i\tau_E)^{-6}=1^{-6}=1$ in the S-transformation
  of $Z[\tau_E]$.  $Z[i]$ is S-invariant. \Lz
\item \textbf{Uniqueness on the imaginary axis}: The fixed points of
  $\mathrm{SL}(2,\ZZ)$ on $\mathbb{H}$ are $\tau=i$ (fixed by $S$)
  and $\tau=e^{2\pi i/3}$ (fixed by $ST$).  For $\tau_E=iR_\psi$ with
  $R_\psi>0$, only $\tau_E=i$ is reachable.  Hence $R_\psi=1$ is the
  \emph{unique} fixed point on the UBT modular parameter space. \Lz
\item \textbf{Stationarity}: $\mathrm{Re}\,S[\Theta]$ is stationary
  under the modular S-flow at $\tau_E=i$ (Proposition~\ref{prop:Smod}).
  \Lz
\end{enumerate}
\textbf{Conclusion}: $R_\psi = 1$ is derived from the algebraic
structure of $Z[\tau_E] = \eta(\tau_E)^{-12}$ and the modular
identity for $\eta$.  Status: \Lo\ conditional on items 1--3 of
Proposition~\ref{prop:Z} (i.e.\ on $\det P=\eta^2$ [\STD],
$N_{\mathrm{eff}}=12$ [\Lo], and one-loop approximation).
\end{theorem}

%──────────────────────────────────────────────────────────────────
\section{Path 1: Casimir Polynomial}
\label{sec:casimir}
%──────────────────────────────────────────────────────────────────

This section provides an independent derivation via dynamics.

The effective action (Casimir energy on $T^3$ with $d=3$,
\cite{ambjorn_wolfram_1983}; $C_{\mathrm{tot}}=N_{\mathrm{eff}}
\cdot 3\zeta(3)/(128\pi^2) > 0$):
\[
  S_{\mathrm{eff}}(R_\psi, T_{\mathrm{kin}})
  = 2\pi R_\psi\,T_{\mathrm{kin}} + \frac{C_{\mathrm{tot}}}{R_\psi^3}.
\]

\begin{lemma}[S-invariance condition — conditional on S-symmetry of $S_{\mathrm{eff}}$]
\label{lem:Sinv}
$S_{\mathrm{eff}}(R) = S_{\mathrm{eff}}(1/R)$ if and only if:
\begin{equation}
  \label{eq:Tkin}
  T_{\mathrm{kin}}(R_\psi) = \frac{C_{\mathrm{tot}}}{2\pi}
  \Bigl(R_\psi^2 + 1 + R_\psi^{-2}\Bigr).
\end{equation}
\textup{(}Factorisation $R^3-R^{-3} = (R-R^{-1})(R^2+1+R^{-2})$. \Lz\textup{)}
\end{lemma}

\begin{theorem}[Polynomial fixing — \Lz\ (polynomial) + \MC\ (S-symmetry)]
\label{thm:poly}
Substituting \eqref{eq:Tkin} into $dS_{\mathrm{eff}}/dR_\psi=0$:
\begin{equation}
  \label{eq:poly}
  \boxed{R_\psi^6 + R_\psi^4 + R_\psi^2 = 3.}
\end{equation}
This has the unique positive real solution $R_\psi = 1$.

\textbf{Proof.}
\emph{Solution}: $1+1+1=3$. \Lz.
\emph{Uniqueness}: $f(u)=u^3+u^2+u-3$ with $u=R_\psi^2>0$.
$f(1)=0$, $f'(u)=3u^2+2u+1>0$ for all $u$ (discriminant $<0$).
Hence $u=1$ is the unique positive root. \Lz.
\emph{$N_{\mathrm{eff}}$ cancels}: $C_{\mathrm{tot}}=N_{\mathrm{eff}}
C_{\mathrm{unit}}$ appears on both sides of the stationarity equation
and divides out.  The polynomial is independent of $N_{\mathrm{eff}}$.
Verified for $N_{\mathrm{eff}}\in\{1,3,6,12,24,48\}$ numerically.

\textbf{Status}: the polynomial \eqref{eq:poly} and its unique
solution are \Lz.  The S-symmetry of $S_{\mathrm{eff}}$ (condition
for Lemma~\ref{lem:Sinv}) is \MC\ — it is not derived from $S[\Theta]$
but is consistent with the result of Path 2 (Theorem~\ref{thm:fixed}).
\end{theorem}

\begin{corollary}[$R_\psi=1$ is a minimum — \Lz]
\label{cor:min}
$d^2S_{\mathrm{eff}}/dR_\psi^2|_{R_\psi=1} = 20C_{\mathrm{tot}} > 0$.
Numerically $\approx 0.685 > 0$. \NUM
\end{corollary}
