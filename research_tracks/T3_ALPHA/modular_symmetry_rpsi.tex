\begin{abstract}
Zkoumáme, zda efektivní akce $S_{\mathrm{eff}}[R_\psi]$ odvozená z $S[\Theta]$
je invariantní pod modulární S-transformací $\tau_E \to -1/\tau_E$ (ekvivalentně
$R_\psi \to 1/R_\psi$, T-dualita na $\psi$-kružnici).  Ukazujeme, že pokud
S-invariance fixuje $T_{\mathrm{kin}}$ skrze podmínku
$S_{\mathrm{eff}}[R] = S_{\mathrm{eff}}[1/R]$, pak stacionarita
$dS_{\mathrm{eff}}/dR = 0$ vede na polynomickou rovnici
$R^6 + R^4 + R^2 = 3$, jejíž jediné kladné reálné řešení je $R = 1$.
Klíčový fakt: $N_{\mathrm{eff}}$ se vyruší — fixace $R_\psi = 1$ je algebraická
vlastnost S-symetrie, nezávislá na $N_{\mathrm{eff}}$.
Výsledek je podmíněný (Result~C): S-invariance $S[\Theta]$ nebyla dosud
odvozena z prvních principů.
\textbf{Alpha NOT DERIVED.}
\end{abstract}

\section{Kontext a motivace}
\label{sec:context}
%──────────────────────────────────────────────────────────────────────────────

Tři otevřené problémy UBT sdílejí společný kořen: poloměr $\psi$-kružnice
$R_\psi$ není odvozen z $S[\Theta]$ bez fenomenologického vstupu.

\begin{enumerate}
  \item \textbf{Alfa (G137-B)}: $B_{\mathrm{phenom}}$ závisí na samoduálním
    bodě $\tau = i$, který nastane pro $R_\psi = 1$.  Proč je $\tau = i$

  \item \textbf{Scale closure (EW-1b)}: $M_{\mathrm{GUT}} = 1/R_\psi^*$;

  \item \textbf{C2-iv}: $\psi$-vinutí a colour fiber jsou odděleny.
    Propojení by mohlo plynout z geometrie $\psi$-kružnice na speciální
    hodnotě $R_\psi$.
\end{enumerate}

\textbf{Hypotéza}: $S[\Theta]$ má modulární symetrii pod $\tau \to -1/\tau$
(inverze komplexního času).  Pokud ano, samoduální bod $\tau = i$ je

\section{Definice modulární transformace v kontextu UBT}
\label{sec:setup}
%──────────────────────────────────────────────────────────────────────────────

\subsection*{Komplexní čas a $\psi$-kružnice}

Komplexní čas $\tau = t + i\psi \in \CC$.  $\psi$-kružnice $S^1_\psi$ má
periodu $2\pi R_\psi$.  Euklidovský modulární parametr:
\[
  \tau_E = iR_\psi \in \mathbb{H},
  \quad t = 0,\; \psi \sim \psi + 2\pi R_\psi.
\]

Standardní modulární transformace na horní polorovině $\mathbb{H}$:
\[
  T:\; \tau_E \to \tau_E + 1, \qquad
  S:\; \tau_E \to -\frac{1}{\tau_E}.
\]
Grupa $\mathrm{SL}(2,\ZZ)$ je generována $T$ a $S$.

\begin{definition}[Modulární S-transformace na UBT komplexním čase]
\label{def:modular_S}
S-transformace na $\psi$-kružnici:
\[
  \tau_E \;\longmapsto\; -\frac{1}{\tau_E},
  \qquad
  iR_\psi \;\longmapsto\; \frac{i}{R_\psi},
\]
ekvivalentně $R_\psi \to 1/R_\psi$ — T-dualita na $\psi$-kružnici
\Lone~(standardní T-dualita~\cite[§8]{polchinski_v1}).

Samoduální bod S-transformace: $S(i) = -1/i = i$, tedy $R_\psi = 1$
je \emph{fixní bod} S-transformace.
\end{definition}

\section{Test modulární invariance kinetického členu}
\label{sec:kinetic}
%──────────────────────────────────────────────────────────────────────────────

Kinetický člen $S[\Theta]$ (z \texttt{appendix\_AA\_theta\_action.tex}):
\[
  S_{\mathrm{kin}} = \int \mathrm{Tr}\!\left[(D_\mu\Theta)^\dagger(D^\mu\Theta)\right]
  d^4x\, d\psi.
\]
Integrace přes $d\psi$ s periodou $2\pi R_\psi$.

\begin{proposition}[Transformace kinetického členu pod $R_\psi \to 1/R_\psi$]
\label{prop:kin_transform}
Pod S-transformací $R_\psi \to 1/R_\psi$:
\begin{itemize}
  \item Perioda: $2\pi R_\psi \to 2\pi/R_\psi$.
  \item KK spektrum: $m_n = n/R_\psi \to n R_\psi$ (inverze hmotností).
  \item Objem: $\int d\psi = 2\pi R_\psi \to 2\pi/R_\psi$.
\end{itemize}
Proto:
\[
  S_{\mathrm{kin}}[\Theta,\, R_\psi]
  \;\xrightarrow{\;R_\psi \,\to\, 1/R_\psi\;}\;
  \frac{1}{R_\psi^2}\, S_{\mathrm{kin}}[\widetilde\Theta,\, 1/R_\psi],
\]
kde $\widetilde\Theta$ je S-duální pole.  Faktor $R_\psi^{-2}$ pochází z
dimenzionální analýzy: $[\Theta] = M^1$ a pole se rescaluje jako
$\widetilde\Theta = R_\psi \cdot \Theta$ \MC.
$S_{\mathrm{kin}}$ je S-invariantní pokud $\widetilde\Theta = R_\psi \cdot \Theta$.
\end{proposition}

%──────────────────────────────────────────────────────────────────────────────
\section{Casimirův člen a podmínka S-invariance $S_{\mathrm{eff}}$}
\label{sec:casimir}
%──────────────────────────────────────────────────────────────────────────────

\subsection*{Casimirův potenciál}

Z \texttt{research\_tracks/EW/rpsi\_from\_action.tex} a standardní Casimirovy
energetiky na $T^3$ s $d = 3$ \Lone~\cite{ambjorn_wolfram_1983}:
\[
  V_{\mathrm{Casimir}}(R_\psi) = \frac{C_{\mathrm{total}}}{R_\psi^3},
  \qquad
  C_{\mathrm{total}} = N_{\mathrm{eff}} \cdot C_{\mathrm{unit}},
  \qquad
  C_{\mathrm{unit}} = \frac{3\,\zeta(3)}{128\pi^2}.
\]
$(C_{\mathrm{total}} > 0$ odpovídá standardní Casimirově repulsi od $R_\psi = 0$.)

\subsection*{Efektivní akce}

\[
  S_{\mathrm{eff}}(R_\psi,\, T_{\mathrm{kin}}) = 2\pi R_\psi \cdot T_{\mathrm{kin}}
  + \frac{C_{\mathrm{total}}}{R_\psi^3}.
\]

\subsection*{S-invariance $S_{\mathrm{eff}}$}

\begin{lemma}[Podmínka S-invariance $S_{\mathrm{eff}}$]
\label{lem:S_inv_condition}
$S_{\mathrm{eff}}[R_\psi,\, T_{\mathrm{kin}}] = S_{\mathrm{eff}}[1/R_\psi,\, T_{\mathrm{kin}}]$
pokud a jen pokud:
\[
  2\pi T_{\mathrm{kin}}\!\left(R_\psi - \frac{1}{R_\psi}\right)
  = V_{\mathrm{Casimir}}\!\left(\frac{1}{R_\psi}\right) - V_{\mathrm{Casimir}}(R_\psi)
  = C_{\mathrm{total}}\!\left(R_\psi^3 - \frac{1}{R_\psi^3}\right).
\]
Pro $R_\psi \neq 1$ tedy:
\begin{equation}
  \label{eq:T_kin_S_inv}
  T_{\mathrm{kin}}(R_\psi) = \frac{C_{\mathrm{total}}}{2\pi}
  \cdot \frac{R_\psi^3 - R_\psi^{-3}}{R_\psi - R_\psi^{-1}}
  = \frac{C_{\mathrm{total}}}{2\pi}
  \left(R_\psi^2 + 1 + \frac{1}{R_\psi^2}\right).
\end{equation}
L'H\^{o}pitalova mez pro $R_\psi \to 1$:
$T_{\mathrm{kin}}(1) = 3C_{\mathrm{total}}/(2\pi)$.
\end{lemma}

\begin{proof}
Přímý výpočet:
$S_{\mathrm{eff}}[R] - S_{\mathrm{eff}}[1/R]
= 2\pi T_{\mathrm{kin}}(R - 1/R) + C_{\mathrm{total}}(1/R^3 - R^3) = 0$
implikuje \eqref{eq:T_kin_S_inv}.
Faktorizace $R^3 - 1/R^3 = (R - 1/R)(R^2 + 1 + 1/R^2)$ \Lzero.
\end{proof}

%──────────────────────────────────────────────────────────────────────────────
\section{Podmínka S-invariance, stacionarita a polynomial}
\label{sec:stationary}
%──────────────────────────────────────────────────────────────────────────────

\subsection*{Stacionární rovnice}

Stacionarita $dS_{\mathrm{eff}}/dR_\psi = 0$:
\[
  \frac{d}{dR_\psi}\!\left[2\pi R_\psi T_{\mathrm{kin}}
  + \frac{C_{\mathrm{total}}}{R_\psi^3}\right] = 0
  \;\;\Longrightarrow\;\;
  2\pi T_{\mathrm{kin}} = \frac{3C_{\mathrm{total}}}{R_\psi^4}.
\]

\subsection*{Klíčové substituce}

Dosadíme $T_{\mathrm{kin}}$ z podmínky S-invariance~\eqref{eq:T_kin_S_inv}:
\[
  \frac{C_{\mathrm{total}}}{2\pi}\cdot 2\pi \cdot
  \left(R_\psi^2 + 1 + \frac{1}{R_\psi^2}\right)
  = \frac{3C_{\mathrm{total}}}{R_\psi^4}.
\]
$C_{\mathrm{total}}$ se vyruší (nezávisí na $R_\psi$):
\[
  R_\psi^2 + 1 + \frac{1}{R_\psi^2} = \frac{3}{R_\psi^4}.
\]
Násobíme $R_\psi^4$:

\begin{equation}
  \label{eq:poly}
  \boxed{R_\psi^6 + R_\psi^4 + R_\psi^2 = 3.}
\end{equation}

\begin{theorem}[Polynomická fixace $R_\psi$ ze S-invariance]
\label{thm:poly_fixation}
Podmíněně na S-invarianci $S_{\mathrm{eff}}$ a Casimirovou formou
$V = C_{\mathrm{total}}/R_\psi^3$, splní $R_\psi$ polynomickou rovnici
\eqref{eq:poly}.  Tato rovnice má jediné kladné reálné řešení $R_\psi = 1$.

Klíčový fakt: $N_{\mathrm{eff}}$ se \emph{vyruší} (obsažen pouze v
$C_{\mathrm{total}}$, které se krátí) — fixace $R_\psi = 1$ je algebraická
vlastnost S-symetrie, nezávislá na $N_{\mathrm{eff}}$.
\end{theorem}

\begin{proof}
\emph{$R_\psi = 1$ je řešení}: $1 + 1 + 1 = 3$ \Lzero.
\emph{Jedinost}: substituce $u = R_\psi^2 > 0$ dává $u^3 + u^2 + u = 3$,
tj.\ $f(u) = u^3 + u^2 + u - 3$.  $f(1) = 0$, $f'(u) = 3u^2 + 2u + 1 > 0$
pro všechna $u$ (diskriminant $< 0$) — $f$ je \emph{ryze rostoucí} na
$(0,\infty)$, tedy $u = 1$ je jediný kladný kořen \Lzero.
\emph{$N_{\mathrm{eff}}$ se vyruší}: $C_{\mathrm{total}} = N_{\mathrm{eff}} C_{\mathrm{unit}}$
se vyskytuje na obou stranách stacionární rovnice a zkrátí se.
Numericky ověřeno pro $N_{\mathrm{eff}} \in \{1, 3, 6, 12, 24, 48\}$
v \texttt{tools/modular\_rpsi\_check.py}.
\end{proof}

\begin{corollary}[R=1 je minimum]
\label{cor:minimum}
$R_\psi = 1$ je \emph{minimum} (ne maximum) $S_{\mathrm{eff}}$ s $T_{\mathrm{kin}}$
ze S-invariance.  Druhá derivace:
\[
  \frac{d^2 S_{\mathrm{eff}}}{dR_\psi^2}\bigg|_{R_\psi=1} = 20\, C_{\mathrm{total}} > 0.
\]
\emph{Výpočet}: $d^2T_{\mathrm{kin}}/dR^2|_{R=1} = 4C_{\mathrm{total}}/\pi$ a
$dT_{\mathrm{kin}}/dR|_{R=1} = 0$ \Lzero.
Tedy $d^2 S/dR^2 = 2\pi \cdot (0 + 4C/\pi) + 12C = 8C + 12C = 20C > 0$.
Numericky: $d^2S/dR^2|_{R=1} = 20 \times 0.03425 \approx 0.685 > 0$. \NUM
\end{corollary}

\subsection*{Rozlišení zdánlivé $N_{\mathrm{eff}} = 1$ nekonzistence}

\begin{remark}[Notační nekonzistence v naivní analýze]
\label{rem:notation}
Naivní analýza, která mísí $C_{\mathrm{unit}}$ (per mode) a $C_{\mathrm{total}}$
(total), dává:
\[
  R^6 + R^4 + R^2 = 3N_{\mathrm{eff}} \;\Rightarrow\; N_{\mathrm{eff}} = 1
  \;\text{(pro $R=1$)}.
\]
Toto je \emph{notační artefakt}: pokud v Lemma~\ref{lem:S_inv_condition}
figuruje $C_{\mathrm{unit}}$ (per-mode), ale stacionarní rovnice používá
$3N_{\mathrm{eff}}C_{\mathrm{unit}}/R^4$, pak se $N_{\mathrm{eff}}$ nekrátí.
Správná analýza s konzistentním $C_{\mathrm{total}}$ dává rovnici~\eqref{eq:poly},
kde $N_{\mathrm{eff}}$ \emph{chybí} — fixace $R=1$ platí pro libovolné
$N_{\mathrm{eff}}$, konkrétně i pro $N_{\mathrm{eff}} = 12$.
\end{remark}

\section{Příspěvek gauge členu}
\label{sec:gauge}
%──────────────────────────────────────────────────────────────────────────────

$S[\Theta]$ obsahuje gaugeový člen
$S_{\mathrm{gauge}} = \int \mathrm{Tr}[F_{\mu\nu}F^{\mu\nu}]\,d^4x\,d\psi$.
Pod S-transformací $R_\psi \to 1/R_\psi$: gauge coupling $g \to g/R_\psi$
(KK rescaling) \cite[§12]{polchinski_v1}.

\begin{lemma}[Efektivní $T_{\mathrm{kin}}^{\mathrm{eff}}$ se gauge příspěvkem]
\label{lem:Tkin_eff}
Zahrneme-li gaugeový člen jako příspěvek k efektivnímu kinetickému členu:
\[
  T_{\mathrm{kin}}^{\mathrm{eff}} = T_{\mathrm{kin}} + \Delta T_{\mathrm{gauge}},
  \quad
  \Delta T_{\mathrm{gauge}} \sim N_{\mathrm{colour}}\,\frac{g^2 M_{\mathrm{KK}}^4}{16\pi^2},
\]
kde $N_{\mathrm{colour}} = 8\,(\text{gluony}) + 3\,(\text{weak}) + 1\,(\text{hypercharge}) = 12$.

\textbf{Pozorování}: $N_{\mathrm{colour}} = N_{\mathrm{eff}} = 12$ \MC.
Tato rovnost může být nenahodná: oba počty reflektují počet fyzikálních
bosonu $\mathrm{SU}(3) \times \mathrm{SU}(2)_L \times \mathrm{U}(1)_Y$
v UBT s $N_{\mathrm{eff}} = 12$ [L1 conditional on SU(3)$\times$SU(2)$\times$U(1)
from UBT].
\end{lemma}

\begin{remark}[Gauge příspěvek a polynom]
\label{rem:gauge_poly}
Přidání $\Delta T_{\mathrm{gauge}}$ modifikuje $T_{\mathrm{kin}}$ v
Lemma~\ref{lem:S_inv_condition}.  Pokud $\Delta T_{\mathrm{gauge}}$ je samo
S-invariantní (tj.\ $\Delta T_{\mathrm{gauge}}(R) = \Delta T_{\mathrm{gauge}}(1/R)/R^2$
po vhodném poli rescaling), pak Theorem~\ref{thm:poly_fixation} zůstává
platný s $C_{\mathrm{total}} \to C_{\mathrm{total}} + C_{\mathrm{gauge}}$.
Podmínka $R = 1$ a polynom~\eqref{eq:poly} jsou robustní vůči takovým
S-invariantním korekcím. Derivace $C_{\mathrm{gauge}}$ z $S[\Theta]$: \OPEN.
\end{remark}

\begin{remark}[C2-iv a $\psi$-kružnice při $R_\psi = 1$]
\label{rem:c2iv}

\begin{itemize}
  \item $S(i) = i$ — $\tau = i$ je fixní bod.

\[
  \psi \;\text{-rotace o }\tfrac{2\pi}{3}
  \;\longleftrightarrow\;
  \text{colour }\ZZ_3\text{-rotace v SU(3)}.
\]
Toto je \emph{kandidátní uzavření C2-iv} \MC\ podmíněné na modulární fixaci
$R_\psi$ a derivaci colour structure z $\psi$-winding (C2-i: \OPEN).
\end{remark}
