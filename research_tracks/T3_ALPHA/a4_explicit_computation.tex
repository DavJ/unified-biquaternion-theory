\section{Setup}
\[
D_{\mathrm{UBT}} = D_{M^4} + i\partial_\psi\otimes\gamma_5,
\qquad
P=D_{\mathrm{UBT}}^2.
\]
Using the Lichnerowicz-type decomposition (\textbf{[STD]}):
\[
P = -\left(g^{\mu\nu}\nabla_\mu\nabla_\nu\otimes 1 + 1\otimes\partial_\psi^2\right)+E,
\]
with endomorphism $E$ and bundle curvature $\Omega_{\mu\nu}$ as in
\texttt{research\_tracks/T3\_ALPHA/seeley\_dewitt\_coefficients.tex}.

\section{Key obstruction from the $M^4$ determinant channel}
For a massive scalar determinant in 4D,
\[
V_{\mathrm{eff}}^{(4D)}(m)\sim m^4\!\left[\ln(m/\mu)-c\right].
\]
With KK mass \(m_n=n/R\), this gives
\[
V_{\mathrm{eff}}^{(4D)}(n)\sim \frac{n^4}{R^4}\ln n,
\]
so the direct \(M^4\) channel cannot produce the target \(n\ln n\) structure.
This is the analytic reason the gap cannot be closed from the 4D determinant alone.

\section{Step 1: $a_4$ structure (context)}
The standard coefficient on product backgrounds is
\[
a_4(P)=\frac{1}{16\pi^2}\int \mathrm{tr}\!\left[
\frac{1}{180}\!\left(5R^2-2R_{\mu\nu}^2+2R_{\mu\nu\rho\sigma}^2\right)
-\frac{1}{6}RE+\frac12E^2+\frac1{12}\Omega_{\mu\nu}^2
\right]\sqrt{g}\,d^5x.
\]
For a product background and factorisable operator data,
\[
a_4(P)|_{M^4\times S^1}
=a_4(D_{M^4}^2)L_{S^1}
+a_2(D_{M^4}^2)a_2(D_{S^1}^2)
+\mathrm{Vol}(M^4)a_4(D_{S^1}^2).
\]

\section{Step 2: T$^2$ winding channel and divisor mechanism}
Take a symmetric compact sector
\(
T^2=S^1_\psi\times S^1_\chi
\)
with operator \(P_{T^2}=-\partial_\psi^2-\partial_\chi^2\).
Its heat kernel is
\[
K_{T^2}(t)=\sum_{m,k\in\ZZ}e^{-t(m^2+k^2)/R^2}
=\vartheta_3\!\left(0\,\middle|\,i\frac{t}{\pi R^2}\right)^2.
\]
Restricting to winding sectors indexed by divisor pairs (\(mk=n\)),
the multiplicity is the divisor function \(D_2(n)=\tau(n)\).
Using the classical asymptotic
\[
\sum_{k\le N}D_2(k)=N\ln N+(2\gamma-1)N+O(\sqrt N),
\]
the cumulative T$^2$ contribution scales as \(N\ln N\).

\begin{proposition}[T$^2$ origin of \(n\ln n\)]
The target potential form
\[
V_{\mathrm{eff}}(n)=n^2-B\,n\ln n
\]
is compatible with a 2D divisor-sum winding channel (T$^2$), while:
\begin{itemize}
  \item the direct 4D determinant channel gives \(n^4\ln n\),
  \item a T$^3$ divisor channel gives \(n(\ln n)^2\) asymptotics.
\end{itemize}
Hence the \(n\ln n\) scaling singles out T$^2$ as the correct asymptotic source.
\end{proposition}

\section{Step 3: coefficient identification problem}
Write schematically
\[
V_{\mathrm{eff}}^{T^2}(n)=n^2-\sum_{d\mid n}F(d;R,\mu),
\]
with \(F\) generated by renormalised determinants on \(T^2\).
The scaling \(n\ln n\) is fixed asymptotically, but the exact closed coefficient map
\[
B \stackrel{?}{=} N_{\mathrm{phases}}^{3/2}\,2^{1/8}\,\vartheta_3(0|i)^{1/4}
\]
still requires an explicit determinant-to-coefficient insertion theorem.

\section{Self-dual \(S^1\) value used in the bridge}
For $P_{S^1}=-\partial_\psi^2$ with radius $R_\psi$:
\[
K_{S^1}(t)=\sum_{n\in\mathbb Z}e^{-tn^2/R_\psi^2}
=\vartheta_3\!\left(0\,\middle|\,i\frac{t}{\pi R_\psi^2}\right).
\]
At the self-dual point $R_\psi\Lambda=1$ with $t_*=R_\psi^2$,
\[
K_{S^1}(t_*)=\vartheta_3(0|i).
\]
Ramanujan's CM value:
\[
\vartheta_3(0|i)=\pi^{1/4}/\Gamma(3/4)\qquad\text{\textbf{[STD]}}.
\]

\section{Target formula and status}
\[
B_{\mathrm{target}}
=12^{3/2}\cdot 2^{1/8}\cdot \vartheta_3(0|i)^{1/4}.
\]
The algebraic/numerical identity is reproducible (see
\texttt{tools/a4\_analytical\_verification.py}), but this is not yet a
first-principles derivation from $S[\Theta]$.
