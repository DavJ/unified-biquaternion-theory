\section*{Scope}
This file states scaling regimes where the reduced equation reproduces established condensate physics and where speculative UBT terms could become testable.

\section{Dimensionless Form}
Let $\mathbf r=L_0\tilde{\mathbf r}$, $t=T_0\tilde t$, $\psi=\Psi_0\tilde\psi$. Then
\begin{equation}
i\partial_{\tilde t}\tilde\psi=\left[-\tilde\nabla^2+\tilde V+u|\tilde\psi|^2+i(\tilde P-\tilde\Gamma)-i\tilde\zeta|\tilde\psi|^2\right]\tilde\psi+\tilde\eta\,\partial_{\tilde t}^2\tilde\psi+\tilde\lambda\,\tilde\nabla^4\tilde\psi+\cdots.
\end{equation}

\section{Established Limit (Condensed-Matter Baseline)}
If
\begin{equation}
|\tilde\eta|\ll1,\qquad |\tilde\lambda|\ll1,
\end{equation}
the dynamics reduces to standard ddGP/CGL classes used in polariton modeling.

\section{Candidate UBT-Correction Window (SPECULATIVE)}
Potentially resolvable corrections require simultaneously:
\begin{equation}
|\tilde\eta|\omega_{\mathrm{obs}}\gtrsim \epsilon_{\mathrm{det}},
\qquad
|\tilde\lambda|k_{\mathrm{obs}}^2\gtrsim \epsilon_{\mathrm{det}},
\end{equation}
where $\epsilon_{\mathrm{det}}$ is the experimental fractional precision.

\section{Mathematical Gaps}
\begin{enumerate}
\item No derived UV-to-IR matching for $(\tilde\eta,\tilde\lambda)$.
\item No error-controlled truncation bound for omitted higher operators.
\item No full identifiability analysis separating UBT corrections from ordinary cavity/model uncertainty.
\end{enumerate}
