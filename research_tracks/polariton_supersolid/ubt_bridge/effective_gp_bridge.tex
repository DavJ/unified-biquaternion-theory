\section*{Scope}
This comparison separates established equations from speculative UBT-correction terms.

\section{Reference Equations (Established)}
\paragraph{Conservative GP}
\begin{equation}
i\hbar\partial_t\psi=\left[-\frac{\hbar^2\nabla^2}{2m}+V+g|\psi|^2\right]\psi.
\end{equation}

\paragraph{Driven-dissipative GP (ddGP)}
\begin{equation}
i\hbar\partial_t\psi=\left[-\frac{\hbar^2\nabla^2}{2m_*}+V+g|\psi|^2+\frac{i\hbar}{2}(R n_R-\Gamma_C)\right]\psi.
\end{equation}

\paragraph{Complex Ginzburg--Landau (CGL) form}
\begin{equation}
\partial_t\psi=(\mu+i\omega_0)\psi-(\beta+i\beta')|\psi|^2\psi+(D_r+iD_i)\nabla^2\psi.
\end{equation}

\section{Candidate UBT-Reduced Equation}
\begin{equation}
i\hbar\partial_t\psi=\left[-\frac{\hbar^2\nabla^2}{2m_*}+V+g|\psi|^2+\frac{i\hbar}{2}(P-\Gamma)-i\hbar\zeta|\psi|^2\right]\psi+\delta\mathcal N_{\mathrm{UBT}}.
\end{equation}
with
\begin{equation}
\delta\mathcal N_{\mathrm{UBT}}=\left[\eta_\tau\partial_t^2+\lambda_k\nabla^4+\sigma\,\nabla\cdot(|\psi|^2\nabla)\right]\psi.
\end{equation}

\section{Mapping Table}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Feature & Standard identification & UBT bridge status \\
\midrule
Kinetic term & $m_*$ in GP/ddGP & Candidate projection from low $\Theta$ mode \\

Gain/loss & $(P-\Gamma)$ or $(Rn_R-\Gamma_C)$ & Assumed via coarse-grained bath \\
Finite-$k$ selector & $\nabla^4$ in CGL-like extensions & SPECULATIVE ($\lambda_k$) \\

\bottomrule
\end{tabular}
\end{center}

\section{Interpretive Boundaries}
\begin{itemize}
\item The GP/ddGP/CGL equations are established condensed-matter tools.
\item The UBT bridge is currently a reduction hypothesis, not a validated derivation.
\item No claim is made that UBT already outperforms standard polariton modeling.
\end{itemize}

\section{Open Mathematical Tasks}
\begin{enumerate}
\item Derive all reduced coefficients from one controlled expansion scheme.
\item Prove consistency of open-system limit with UBT complex-time dynamics.
\item Establish renormalization-group relevance/irrelevance of $\delta\mathcal N_{\mathrm{UBT}}$.
\end{enumerate}
