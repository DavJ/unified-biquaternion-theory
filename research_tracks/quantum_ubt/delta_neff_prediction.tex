\begin{abstract}
The UBT Scherk--Schwarz twist on $S^1_\psi$ generates $N_{\rm eff}^{(KK)}=12$
charged Kaluza--Klein modes with masses $m_n \sim n\,M_{\rm Pl}$.
These modes decouple from the thermal bath at $T_{\rm dec} \sim M_{\rm Pl}$
and produce a relic contribution to the CMB effective number of relativistic
degrees of freedom. Using the $g_*$-dependent relic formula, we find:
$\Delta N_{\rm eff}^{(\rm total)}\approx 1.569$ for $g_*=106.75$ (SM only; tension),
while $\Delta N_{\rm eff}^{(\rm total)}\approx 0.247$ for $g_*=427$ (SM+KK; consistent).
\textbf{Alpha: NOT DERIVED.}
\end{abstract}

\section{Setup: UBT KK Modes and Decoupling}
\label{sec:setup}
%──────────────────────────────────────────────────────────────────────────────

From \texttt{research\_tracks/quantum\_ubt/su2\_twist\_neff12.tex}~[L1],
the Scherk--Schwarz boundary condition
\[
  \Theta(\psi + 2\pi R_\psi) = \sigma_3\,\Theta\,\sigma_3
\]
on $S^1_\psi$ with $R_\psi \sim \ell_{\rm Pl}$ generates $N_{\rm eff}^{(KK)}=12$
effective charged bosonic KK modes in the one-loop effective potential.

The $n$-th KK excitation has mass
\[
  m_n = \frac{n}{R_\psi} \sim n\,M_{\rm Pl},
  \qquad n = 1,2,3,\ldots
\]
Since $m_1 \sim M_{\rm Pl} \gg T_{\rm BBN} \sim 1\,\mathrm{MeV}$,
all KK modes are non-relativistic at BBN and have decoupled
from the thermal bath at $T_{\rm dec} \sim M_{\rm Pl}$.

\section{Standard Relic Formula}
\label{sec:formula}
%──────────────────────────────────────────────────────────────────────────────

For a species with $g_X$ real bosonic degrees of freedom that decoupled
from the thermal bath in radiation domination at temperature $T_{\rm dec}$,
the contribution to $N_{\rm eff}$ today is
\begin{equation}
  \label{eq:delta_neff_standard}
  \Delta N_{\rm eff} = g_X\,\frac{43}{4}\left(\frac{g_{*s}(T_0)}{g_*(T_{\rm dec})}\right)^{\!4/3},
\end{equation}
where $g_*(T_{\rm dec})$ is the effective relativistic d.o.f.\ at decoupling
and $g_{*s}(T_0)=3.909$ is the present entropy d.o.f.\ factor.

\subsection*{Derivation of equation~(\ref{eq:delta_neff_standard})}

After decoupling the species temperature $T_X$ evolves as
$T_X \propto 1/a$ (no further heating).  Entropy conservation
in the SM thermal bath gives
\[
  \frac{T_X}{T_\gamma} = \left(\frac{g_{*s}(T_0)}{g_{*s}(T_{\rm dec})}\right)^{1/3}
\]
where $g_{*s}(T_0)=3.909$.
Today $T_\nu = (4/11)^{1/3} T_\gamma$, so
\[
  \frac{T_X}{T_\nu} = \left(\frac{g_{*s}(T_0)}{g_*(T_{\rm dec})}\right)^{1/3},
\]
and $\Delta N_{\rm eff} = g_X(T_X/T_\nu)^4$ yields~(\ref{eq:delta_neff_standard}).

\subsection{Calculation}

Per KK mode ($g_X = 1$, $g_*=106.75$):
\[
  \Delta N_{\rm eff}^{(\rm mode)}
  = \frac{43}{4}\left(\frac{3.909}{106.75}\right)^{4/3}
  \approx \mathbf{0.1308}.
\]

Total from $N_{\rm eff}^{(KK)} = 12$ modes:
\[
  \Delta N_{\rm eff}^{(\rm total)} = 12 \times 0.1308 \approx \mathbf{1.569}.
\]

g* scan (12 modes):
\[
\begin{array}{lcl}
g_*=106.75 &:& \Delta N_{\rm eff}^{(\rm total)}\approx 1.569 \quad (\text{tension}) \\
g_*=427 &:& \Delta N_{\rm eff}^{(\rm total)}\approx 0.247 \quad (\text{consistent}) \\
g_*=1000 &:& \Delta N_{\rm eff}^{(\rm total)}\approx 0.079 \quad (\text{consistent})
\end{array}
\]

Numerical verification: \texttt{tools/delta\_neff\_calc.py} confirms these values
and gives $g_*\sim 389$ as the consistency threshold for
$\Delta N_{\rm eff}^{(\rm total)}<0.28$.

\subsection{Comparison with quoted value}

The value $\Delta N_{\rm eff} \approx 0.131$ corresponds to the \emph{per-mode}
contribution for $g_*=106.75$. The total from all 12 modes is then
$\approx 1.569$.

\section{Physical interpretation of $g_*(T_{\rm dec})$}
\label{sec:gstar_prop}
%──────────────────────────────────────────────────────────────────────────────

\begin{proposition}[Planck-scale decoupling implies large $g_*$]
KK modes have masses $m_n \sim n\,M_{\rm Pl}$ and gravitational coupling.
Using $\Gamma \sim G_N T^3 m \sim T^3/M_{\rm Pl}^2$ and $H\sim T^2/M_{\rm Pl}$,
the decoupling condition $\Gamma=H$ gives $T_{\rm dec}\sim M_{\rm Pl}$.
At this scale, $g_*(T_{\rm dec})$ includes SM plus KK tower states, so
$g_*(T_{\rm dec})\gg 106.75$ is physically expected.
A conservative lower estimate is $g_*\ge 118.75$ (SM + 12 KK), while
realistic Planck-spectrum estimates are $g_*\sim 427$--$1000$.
\end{proposition}

\section{Observational Consistency Note}
\label{sec:consistency}
%──────────────────────────────────────────────────────────────────────────────

\begin{warnbox}
For $g_*=106.75$, the \emph{total} contribution
$\Delta N_{\rm eff}^{(\rm total)} \approx 1.569$ is in tension with
Planck+ACT. For Planck-scale decoupling with large-$g_*$ spectra
($g_*\sim 427$--$1000$), the prediction shifts to
$\Delta N_{\rm eff}^{(\rm total)} \sim 0.25$--$0.08$, consistent with
current bounds and testable by CMB-S4.
\end{warnbox}

Several mitigating factors require investigation:
\begin{enumerate}
\item \textbf{Mode stability}: Not all 12 KK modes may be stable on cosmological
  time-scales.  Short-lived KK modes decay and do not contribute as
  relic radiation.  A decay-rate analysis is needed.
\item \textbf{g* counting}: If $g_*(T_{\rm dec})$ is significantly larger than
  106.75 (e.g.\ including KK tower and Planck-spectrum states),
  the total contribution decreases. For $g_* = 427$: total $\approx 0.247$,
  and for $g_*=1000$: total $\approx 0.079$.
\item \textbf{Non-equilibrium}: The KK modes may never have been in thermal
  equilibrium if their coupling to the SM thermal bath is suppressed
  by powers of $M_{\rm Pl}^{-1}$.
\end{enumerate}
Resolution of these issues is tracked as~\textbf{[OPEN]} in this file.
