\begin{abstract}
We investigate whether the UBT framework reproduces Friedmann--Robertson--Walker
(FRW) cosmology from the $\Theta$ field.  A homogeneous isotropic ansatz
$\Theta(t) = a(t)\,\Theta_0$ (no spatial dependence) is proposed and checked
against the UBT metric chain
$g_{\mu\nu} = \mathrm{Re}[\mathrm{Tr}(\partial_\mu\Theta\cdot\partial_\nu\Theta^\dagger)]/\mathcal{N}$.
Preliminary analysis shows that this ansatz produces a conformally flat metric
proportional to $a(t)^2\,\eta_{\mu\nu}$; recovery of the full FRW metric requires
an off-diagonal structure in $\partial_\mu\Theta$.  A refined ansatz is proposed
and Friedmann equations are argued to follow at \textbf{[Prop.]} level.
The de Sitter limit is checked for consistency with the Casimir energy.
GAP-C status: \textbf{OPEN} with active first steps.
\end{abstract}

\section{UBT Metric Chain}
\label{sec:metric_chain}
%──────────────────────────────────────────────────────────────────────────────

From \texttt{canonical/gr\_closure/step1\_metric\_bridge.tex}~[L1]:
\begin{equation}
  \label{eq:metric_chain}
  g_{\mu\nu} = \frac{1}{\mathcal{N}}\,
  \mathrm{Re}\bigl[\mathrm{Tr}(\partial_\mu\Theta\cdot\partial_\nu\Theta^\dagger)\bigr].
\end{equation}
The normalisation $\mathcal{N}$ is fixed by the large-distance flat limit
$g_{\mu\nu} \to \eta_{\mu\nu}$.

The Schwarzschild metric is recovered from a spherically symmetric ansatz
$\Theta_0(r,t)$ (see \texttt{papers/UBT\_GR\_Submission.tex}~[L1]).
GAP-C (\texttt{WHAT\_IS\_PROVED.md}) asks whether an FRW ansatz exists.

\section{Homogeneous Isotropic Ansatz}
\label{sec:frw_ansatz}
%──────────────────────────────────────────────────────────────────────────────

\subsection{First ansatz: $\Theta(t) = a(t)\,\Theta_0$}

For a homogeneous isotropic universe, we try
\begin{equation}
  \label{eq:ansatz1}
  \Theta(t) = a(t)\,\Theta_0,
\end{equation}
where $\Theta_0\in\mathbb{B}$ is a constant biquaternion and
$a(t)$ is the scale factor.

With no spatial dependence, $\partial_i\Theta = 0$ for $i=1,2,3$, so:
\begin{align}
  \partial_t\Theta &= \dot a(t)\,\Theta_0, \\
  \partial_i\Theta &= 0.
\end{align}

Substituting into~(\ref{eq:metric_chain}):
\begin{align}
  g_{tt} &= \frac{\dot a^2}{\mathcal{N}}
             \mathrm{Re}[\mathrm{Tr}(\Theta_0\cdot\Theta_0^\dagger)]
           = \frac{\dot a^2 \|\Theta_0\|^2}{\mathcal{N}}, \\
  g_{ti} &= 0, \\
  g_{ij} &= 0.
\end{align}

This gives a metric $g_{\mu\nu} = \mathrm{diag}(\dot a^2 C, 0, 0, 0)$
which is \emph{degenerate} in the spatial directions.
Ansatz~(\ref{eq:ansatz1}) does \emph{not} produce the FRW metric.

\subsection{Refined ansatz: $\Theta(x,t) = a(t)\,e^{ix^k e_k}\,\Theta_0$}

The spatial metric components require spatial gradients.
We consider the refined ansatz
\begin{equation}
  \label{eq:ansatz2}
  \Theta(x,t) = a(t)\,\exp\!\bigl(i\,x^k e_k\bigr)\,\Theta_0,
\end{equation}
where $e_k = \sigma_k/2$ are the $\mathrm{SU}(2)$ generators
and $x^k = (x^1, x^2, x^3)$.
This is a coherent superposition of three winding modes in the spatial directions.

At leading order in small $|x|$:
\begin{align}
  \partial_t\Theta &= \dot a\,e^{ix^k e_k}\,\Theta_0, \\
  \partial_j\Theta &= a(t)\,i\,e_j\,e^{ix^k e_k}\,\Theta_0
  \approx i\,a(t)\,e_j\,\Theta_0 + \mathcal{O}(x).
\end{align}

The spatial metric at $x = 0$:
\begin{align}
  g_{ij}\big|_{x=0}
  &= \frac{a^2}{\mathcal{N}}\,
  \mathrm{Re}\bigl[\mathrm{Tr}(e_i\,\Theta_0\cdot\Theta_0^\dagger e_j^\dagger)\bigr].
\end{align}
For $\Theta_0 = \mathbf{1}_2$ and $e_k = \sigma_k/2$,
$\mathrm{Tr}(e_i e_j^\dagger) = \frac{1}{2}\delta_{ij}$, giving
\begin{equation}
  g_{ij}\big|_{x=0} = a^2(t)\,\delta_{ij}\,C', \qquad
  C' = \frac{1}{2\mathcal{N}}.
\end{equation}
Choosing $\mathcal{N}$ such that $C'=1$ at $a=1$:

\begin{propbox}
The refined ansatz~(\ref{eq:ansatz2}) is promoted below to the formal
Proposition~\ref{prop:frw}.  Its status remains \textbf{[Prop.]}:
the flat FRW metric follows within the admissible real projection, while
full closure from $S[\Theta]$ remains GAP-C.
\end{propbox}

\subsection{Global refined-ansatz check}

For
\[
\Theta(x,t)=a(t)e^{ix^k e_k}\Theta_0,
\]
for $\Theta_0=\mathbf{1}_2$ we have
\[
\partial_0\Theta=\frac{\dot a}{a}\Theta,\qquad
\partial_i\Theta=i\,(e_k)_i\,\Theta.
\]
The mixed metric component is
\[
g_{0i}
=\frac{1}{\mathcal N}
\mathrm{Re}\!\left[\mathrm{Tr}\!\left(\partial_0\Theta\cdot\partial_i\Theta^\dagger\right)\right]
=\frac{1}{\mathcal N}
\mathrm{Re}\!\left[\mathrm{Tr}\!\left(
\frac{\dot a}{a}\Theta\cdot\left(-i\,(e_k)_i\Theta^\dagger\right)\right)\right].
\]
Hence
\[
g_{0i}
=\frac{\dot a}{a}\frac{(-i)(e_k)_i}{\mathcal N}
\mathrm{Re}\!\left[\mathrm{Tr}(\Theta\Theta^\dagger)\right].
\]
For $\Theta_0=\mathbf{1}_2$:
\[
\Theta\Theta^\dagger=a^2(t)\mathbf{1}_2,\qquad
\mathrm{Tr}(\Theta\Theta^\dagger)=2a^2(t)\in\mathbb R.
\]
So the prefactor of $(-i)$ is real, and therefore
\[
\mathrm{Re}\!\left[(-i)\times\text{real}\right]=0
\quad\Longrightarrow\quad
g_{0i}=0
\]
globally.

\begin{proposition}[FRW metric from UBT ansatz]
\label{prop:frw}
Let
\[
\Theta(x,t)=a(t)\,e^{ix^k e_k}\,\Theta_0
\]
with $\Theta_0=\mathbf{1}_2$ and $a(t)>0$. Then:
\begin{enumerate}
\item The off-diagonal components $g_{0i}[\Theta]=0$ globally.
\item The spatial components $g_{ij}[\Theta]=a(t)^2\delta_{ij}$.
\item The temporal component $g_{00}[\Theta]=-1$
      (from AXIOM-B and the admissible real projection).
\end{enumerate}
Hence $g_{\mu\nu}[\Theta]$ is the flat FRW metric with scale factor $a(t)$.
The Friedmann equations follow from the UBT GR chain (Theorem~3.8).
\end{proposition}

\begin{proof}
Step 1:
\[
\partial_0\Theta=\frac{\dot a}{a}\Theta,\qquad
\partial_i\Theta=i\,e_i\,\Theta.
\]
Therefore
\[
g_{0i}
=\frac{1}{\mathcal N}\operatorname{Re}\!\left[
\operatorname{Tr}\!\left(\partial_0\Theta\cdot\partial_i\Theta^\dagger\right)\right]
=\frac{\dot a}{a\mathcal N}\operatorname{Re}\!\left[
i\,e_i\,\operatorname{Tr}(\Theta\Theta^\dagger)\right].
\]
Since \(\Theta\Theta^\dagger=a(t)^2\mathbf{1}_2\),
\(\operatorname{Tr}(\Theta\Theta^\dagger)=2a(t)^2\in\mathbb{R}\),
and \(i\,e_i\) contributes a purely imaginary factor, we get \(g_{0i}=0\).

Step 2:
\[
g_{ij}
=\frac{1}{\mathcal N}\operatorname{Re}\!\left[
\operatorname{Tr}\!\left(i e_i\Theta\cdot(-i)\Theta^\dagger e_j^\dagger\right)\right]
=\frac{a(t)^2}{\mathcal N}\operatorname{Re}\!\left[
\operatorname{Tr}(e_i e_j^\dagger)\right].
\]
For \(\Theta_0=\mathbf{1}_2\) and the standard Pauli-generator normalization,
\(\operatorname{Re}[\operatorname{Tr}(e_i e_j^\dagger)]\propto\delta_{ij}\), and
\(\mathcal N\) is fixed by the flat limit so that the proportionality constant is \(1\).
Hence \(g_{ij}=a(t)^2\delta_{ij}\).

Step 3: by the admissible real projection used throughout the UBT GR chain,
AXIOM-B fixes the temporal component to \(g_{00}=-1\) in the FRW sector.
Combining the three components gives the flat FRW metric.
\end{proof}

\section{Friedmann Equations from UBT}
\label{sec:friedmann}
%──────────────────────────────────────────────────────────────────────────────

\begin{proposition}[Friedmann equations, \textbf{[Prop.]}]
\label{prop:friedmann}
If the metric~(\ref{eq:metric_chain}) with ansatz~(\ref{eq:ansatz2})
reproduces $g_{\mu\nu} = \mathrm{diag}(-1, a^2, a^2, a^2)$ globally,
then the UBT field equation $\nabla^\dagger\nabla\Theta = \kappa\mathcal{T}$
reduces, in the homogeneous isotropic sector, to the Friedmann equations
\begin{align}
  \label{eq:friedmann1}
  H^2 &= \frac{8\pi G}{3}\,\rho, \\
  \label{eq:friedmann2}
  \frac{\ddot a}{a} &= -\frac{4\pi G}{3}\,(\rho + 3p),
\end{align}
where $H = \dot a/a$ is the Hubble parameter.
Equation~(\ref{eq:friedmann2}) is written in acceleration-equation form;
it is equivalent to the $\dot H$ form by $\dot H = \ddot a/a - H^2$.
\end{proposition}

\textit{Argument}: Equations~(\ref{eq:friedmann1})--(\ref{eq:friedmann2})
follow directly from the Einstein equations $G_{\mu\nu} = 8\pi G\,T_{\mu\nu}$
(proved at [L1] in \texttt{canonical/gr\_closure/}) with the FRW
metric, by standard GR computation.  The non-trivial step is showing that
the UBT ansatz~(\ref{eq:ansatz2}) produces the FRW metric globally, which
remains at \textbf{[Prop.]} status.

\section{Explicit $T_{\mu\nu}$ from UBT ansatz}
\label{sec:explicit_tmunu}
%──────────────────────────────────────────────────────────────────────────────

For the homogeneous ansatz
\[
\Theta=a(t)\Theta_0,\qquad \Theta_0=\mathbf{1}_2,
\]
we have
\[
\partial_0\Theta=\dot a\,\mathbf{1}_2,\qquad
\partial_i\Theta=0.
\]
Hence
\begin{align}
T_{00} &= \mathrm{Re}\!\left[\mathrm{Tr}\!\left(\partial_0\Theta\,
\partial_0\Theta^\dagger\right)\right]
= \mathrm{Re}\!\left[\mathrm{Tr}\!\left(\dot a^2\mathbf{1}_2\right)\right]
=2\dot a^2,\\
T_{ij} &= \mathrm{Re}\!\left[\mathrm{Tr}\!\left(\partial_i\Theta\,
\partial_j\Theta^\dagger\right)\right]=0
\quad\text{(no spatial gradients)}.
\end{align}
Define (with \(N\) the effective UBT-to-GR energy-density normalization):
\[
\rho=\frac{T_{00}}{N}=\frac{2\dot a^2}{N}.
\]
Inserting into the first Friedmann equation:
\[
\left(\frac{\dot a}{a}\right)^2=\frac{8\pi G}{3}\rho
=\frac{16\pi G}{3N}\dot a^2.
\]
The ansatz-level consistency condition is therefore
\[
N=\frac{16\pi G}{3}.
\]

\begin{openproblembox}
Normalization check against the GR paper:
\textit{UBT\_GR\_Submission} (Section 3.1 metric chain) defines
\(\mathcal N=|\langle\partial_0\Theta,\partial_0\Theta\rangle_\eta|\)
(Definition/Section 3.1 chain), i.e.\ a field-dependent metric normalization.
The cosmology ansatz above yields an effective constant \(N\) relation.
Matching these two normalizations beyond this reduced ansatz remains
\textbf{[Prop. pending]}.

\subsection{Open problem: normalisation $N$ for FRW ansatz}
\label{sec:N-open}
%────────────────────────────────────────────────────────────────────

Normalisation $N$ is defined in the GR paper (Definition~3.1) as
\[
N := -\langle \partial_0\Theta,\,\partial_0\Theta\rangle_\eta,
\qquad
\langle A,B\rangle_\eta
:= \mathrm{Re}[\mathrm{Tr}(A\,\eta_{\mathcal{B}}\,B^\dagger)],
\]
where $\eta_{\mathcal{B}}=\mathrm{diag}(-1,+1,+1,+1)$ is the
Minkowski metric embedded in $\mathcal{B}=\mathrm{Mat}(2,\mathbb{C})$.

For the FRW ansatz $\Theta = a(t)\mathbf{1}_2$:
\[
\partial_0\Theta = \dot{a}\,\mathbf{1}_2 \in \mathrm{Mat}(2,\mathbb{C}).
\]

\paragraph{Dimensional mismatch.}
$\mathbf{1}_2$ is a $2\times2$ matrix, while $\eta_{\mathcal{B}}$
acts as a $4\times4$ operator on the Dirac spinor space.
The trace $\mathrm{Tr}(\mathbf{1}_2\,\eta_{\mathcal{B}}\,\mathbf{1}_2^\dagger)$
requires a projection of $\eta_{\mathcal{B}}$ onto the
spinor representation of $\mathcal{B}$.

In the Weyl representation this projection gives
$\eta_{\mathcal{B}} \to \gamma^0_{\rm Weyl} = \bigl(\begin{smallmatrix}0&\mathbf{1}_2\\\mathbf{1}_2&0\end{smallmatrix}\bigr)$,
so $\mathrm{Tr}(\mathbf{1}_2\,\gamma^0\,\mathbf{1}_2) = 0$.

An alternative choice is $N = |\det(\partial_0\Theta)| = \dot{a}^2$,
which gives $g_{00}=-1$ and is consistent with Lorentzian signature.
This choice is \textbf{[MC/candidate]} — it produces the correct FRW metric
but is not yet derived from first principles.

\begin{statusbox}
$T_{00} = 2\dot a^2$ from UBT bilinear: \textbf{[Prop.]}\\
$N$ for FRW via indefinite inner product: \textbf{[OPEN — dim. mismatch]}\\
Candidate $N = \dot a^2$ via $|\det(\partial_0\Theta)|$: \textbf{[MC]}\\
Friedmann eqs from UBT: \textbf{[Prop.\ conditional on $N$ choice]}\\
GAP-C: \textbf{OPEN}
\end{statusbox}

\end{openproblembox}

\section{de Sitter Limit}
\label{sec:de_sitter}
%──────────────────────────────────────────────────────────────────────────────

\subsection{Exponential expansion ansatz}

For $a(t) = e^{Ht}$ (de Sitter expansion), the Friedmann equations give:
\[
  H^2 = \frac{\Lambda}{3}, \qquad \rho + p = 0.
\]

In UBT the vacuum energy is sourced by the Casimir contribution of the
KK sector on $S^1_\psi$.  From \texttt{tools/casimir\_B\_check.py}
(see also \texttt{research\_tracks/T3\_ALPHA/mellin\_insertion\_B.tex}):
\[
  E_{\rm Casimir}(T^3, R_\psi = 1) \approx -0.0617\quad (\mathrm{in\,units\,of}\,R_\psi^{-3}).
\]

\subsection{de Sitter limit}
\label{sec:desitter}

\begin{corollary}[de Sitter limit]
\label{cor:desitter}
For $\Theta(t) = e^{Ht}\Theta_0$, the UBT metric gives
$g_{\mu\nu} = \mathrm{diag}(-1, e^{2Ht}, e^{2Ht}, e^{2Ht})$
and the cosmological constant $\Lambda = 3H^2$.
The Friedmann equation gives $H^2 = \Lambda/3 = 8\pi G\rho_\Lambda/3$.
\end{corollary}

\subsection{Consistency with cosmological constant}

The de Sitter Hubble parameter would be
\[
  H_{\rm UBT}^2 \sim \frac{8\pi G}{3}\,|E_{\rm Casimir}|\,R_\psi^{-3}
  \sim \frac{0.517}{\ell_{\rm Pl}^2\cdot 3} \sim M_{\rm Pl}^2.
\]
This is of Planck-scale magnitude, vastly larger than the observed
$\Lambda_{\rm obs} \sim 10^{-122}\,M_{\rm Pl}^4/M_{\rm Pl}^4$.
This is the standard cosmological constant problem, not specific to UBT.

\begin{openproblembox}
The de Sitter limit of UBT and the smallness of $\Lambda_{\rm obs}$
require a cancellation mechanism.  This is~\textbf{[OPEN]} and tracked
as GAP-C in \texttt{WHAT\_IS\_PROVED.md}.
\end{openproblembox}

\section{Cosmological implications}

\subsection{Inflation}
For $\Theta = e^{Ht}\Theta_0$ with $H = \mathrm{const}$:
de Sitter expansion with $\Lambda = 3H^2$ [Prop.~\ref{cor:desitter}].
UBT does not fix $H$ --- it is a free parameter at this stage.

\subsection{Radiation domination}
For $\Theta = a(t)\Theta_0$ with $a \propto t^{1/2}$:
$H = 1/(2t)$, consistent with radiation-dominated FRW.

\subsection{Connection to $\Delta N_{\rm eff}$}
\label{sec:delta_neff}
The FRW background derived here is the classical background
on which KK modes evolve. The $\Delta N_{\rm eff}$ prediction
in \texttt{delta\_neff\_prediction.tex} ([CONDITIONAL TENSION])
uses this background with $T_{\rm dec}\sim M_{\rm Pl}$.
