\section{Charge Quantisation $Q\in\mathbb{Z}$ — \Lz}
\label{sec:charge}
%──────────────────────────────────────────────────────────────────

\begin{theorem}[$Q\in\mathbb{Z}$ from periodicity — \Lz]
\label{thm:charge}
The U(1)$_{\rm EM}$ charge of any state in the $\psi$-winding sector
is an integer: $Q\in\mathbb{Z}$.
\end{theorem}

\begin{proof}
The field $\Theta$ satisfies the periodicity condition
$\Theta(\psi+2\pi R_\psi) = \sigma_3\Theta(\psi)\sigma_3$ [\Lo, SS BC].
Under a U(1)$_{\rm EM}$ gauge transformation $A_\psi\to A_\psi+\partial_\psi\chi$
with $\chi(\psi)=Q\psi/(2\pi R_\psi)$, the field transforms as
$\Theta\to e^{iQ}\Theta$.
For single-valuedness around $S^1_\psi$ of period $2\pi R_\psi$:
\[
  e^{iQ\cdot2\pi R_\psi/(2\pi R_\psi)} = e^{iQ} = 1
  \;\Longrightarrow\; Q\in 2\pi\mathbb{Z}.
\]
With the standard normalisation $e_0=1/(2\pi)$ for the fundamental
U(1)$_{\rm EM}$ charge quantum [\STD], this gives $Q\in\mathbb{Z}$. \Lz
\end{proof}

\begin{remark}
Previously labelled \Lc\ (conditional on self-dual point).
The proof uses only the periodicity of $\Theta$ on $S^1_\psi$ [\Lo,
SS BC] and the standard charge normalisation [\STD]. It does not
require the self-dual condition $R_\psi=1$. Hence upgraded to \Lz.
\end{remark}
