\section{Step 1: Casimir input with \(N_{\mathrm{eff}}=12\)}
Conditionally on SU(2)-twist closure (
\texttt{research\_tracks/quantum\_ubt/su2\_twist\_neff12.tex}), use
\[
V_{\mathrm{Casimir}}(R)=-N_{\mathrm{eff}}\frac{\zeta(-1)}{4R^2}\times \mathcal{N}_{\mathrm{Cas}},
\qquad \zeta(-1)=-\frac{1}{12}.
\]
Thus
\[
V_{\mathrm{Casimir}}(R)=\frac{1}{4R^2}\,\mathcal{N}_{\mathrm{Cas}}
\quad\text{for }N_{\mathrm{eff}}=12.
\]

\section{Step 2: stationarity equation}
From
\[
S_{\mathrm{eff}}=\int d^4x\left[\frac{R}{2\kappa}+(2\pi R_\psi)\,T_{\mathrm{kin}}+V_{\mathrm{Casimir}}(R_\psi)\right],
\quad
T_{\mathrm{kin}}:=\mathrm{Tr}[(D_\mu\Theta)^\dagger(D^\mu\Theta)],
\]
stationarity gives
\[
\frac{dS_{\mathrm{eff}}}{dR_\psi}=2\pi T_{\mathrm{kin}}-\frac{dV_{\mathrm{Casimir}}}{dR_\psi}=0,
\]
so
\[
R_\psi^*=f(T_{\mathrm{kin}},N_{\mathrm{eff}},\mathcal{N}_{\mathrm{Cas}}).
\]
Without fixed \(T_{\mathrm{kin}}\)-normalization, this is not unique.

\section{Step 3: implication for \(M_{\mathrm{GUT}}=1/R_\psi^*\)}
\[
M_{\mathrm{GUT}}=\frac{1}{R_\psi^*}\sim M_{\mathrm{Pl}}\left(\frac{T_{\mathrm{kin}}}{M_{\mathrm{Pl}}^4}\right)^{1/2}.
\]
If \(T_{\mathrm{kin}}\sim M_{\mathrm{GUT}}^4\), one gets \(M_{\mathrm{GUT}}\sim M_{\mathrm{Pl}}\) (too high).
With loop-suppressed input
\(T_{\mathrm{kin}}\sim g^2M_{\mathrm{GUT}}^4/(16\pi^2)\),
\[
M_{\mathrm{GUT}}\sim \frac{M_{\mathrm{Pl}}}{\sqrt{16\pi^2/g^2}},
\]
which is compatible with \(\sim2\times10^{16}\,\mathrm{GeV}\) for \(g\sim0.6\).
This remains conditional on normalization closure.

\section{Step 4: missing condition for uniqueness}
If \(R_\psi^*\) is not unique, an extra closure condition is needed, e.g.
\begin{itemize}
  \item self-consistency input for \(T_{\mathrm{kin}}\),
  \item minimal sensitivity / stability condition \(\partial^2 S_{\mathrm{eff}}/\partial R_\psi^2>0\).
\end{itemize}
