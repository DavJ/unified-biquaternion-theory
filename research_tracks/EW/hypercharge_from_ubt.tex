\begin{abstract}
We derive the Standard Model hypercharge assignments
$Y(Q_L)=\tfrac{1}{6}$, $Y(u_R)=\tfrac{2}{3}$, $Y(d_R)=-\tfrac{1}{3}$,
$Y(L)=-\tfrac{1}{2}$, $Y(e_R)=-1$, $Y(\nu_R)=0$
from UBT $\psi$-winding topology under the assumption that $SU(2)_R$ is
decoupled (OP-S4 [L1 conditional]).
The key relation $Y = (B{-}L)/2$ is established at
\textbf{[L1 conditional on OP-S4 + SU(3) colour structure from UBT]}.
Sub-gap C2-i ($B_q=1/3$ from SU(3) colour-singlet constraint) is closed at
[L1 conditional on SU(3) colour structure from UBT] — see \S\ref{sec:c2i},
Lemma~\ref{lem:Bq_from_su3}.
A separate residual sub-gap (C2-ii) concerns whether $U(1)_B$ arises
from $\psi$-winding topology; this does not block any current paper.
\end{abstract}

\section{The Hypercharge Formula $Y = (B-L)/2$}
\label{sec:hypercharge_formula}
%─────────────────────────────────────────────────────────────────────────────

\begin{lemma}[Hypercharge from $\psi$-winding and OP-S4,
              \textbf{[L1 conditional on OP-S4 + SU(3) colour structure from UBT]}]
\label{lem:hypercharge_formula}
Assume:
\begin{enumerate}
  \item[(A)] $SU(2)_R$ is decoupled at the observational energy scale
             (OP-S4 [L1 conditional]), so the right-handed weak isospin
             satisfies $T_{3R}=0$ for every observed fermion.
  \item[(B)] The $U(1)_Y$ charge is embedded in the extended electroweak
             group as
             \begin{equation}
               Y \;=\; T_{3R} + \frac{B{-}L}{2},
               \label{eq:gell_mann_nishijima}
             \end{equation}
             where $B$ is baryon number and $L$ is lepton number.
  \item[(C)] Baryon number is assigned by the SU(3) colour-singlet constraint
             (Lemma~\ref{lem:Bq_from_su3}): quarks carry $B_q=1/3$
             [L1 conditional on SU(3) colour structure from UBT] and
             leptons carry $L=1$.
\end{enumerate}
Then, under assumption (A), equation \eqref{eq:gell_mann_nishijima} reduces to
\begin{equation}
  \boxed{Y \;=\; \frac{B{-}L}{2},}
  \label{eq:Y_BL}
\end{equation}
and all six SM hypercharge assignments follow algebraically.
\end{lemma}

\begin{proof}
Set $T_{3R}=0$ in \eqref{eq:gell_mann_nishijima}.
The baryon-minus-lepton charge $B{-}L$ and resulting hypercharge
$Y = T_{3R} + (B{-}L)/2$ for each representation are:
\begin{center}
\begin{tabular}{lccccc}
\toprule
Representation & $B$ & $L$ & $B{-}L$ & $T_{3R}$ (pre-OP-S4) & $Y$ \\
\midrule
$Q_L = (u_L, d_L)^T$ & $1/3$ & $0$ & $1/3$ & $0$ & $1/6$ \\
$u_R$               & $1/3$ & $0$ & $1/3$ & $+1/2$ & $2/3$ \\
$d_R$               & $1/3$ & $0$ & $1/3$ & $-1/2$ & $-1/3$ \\
$L   = (\nu_L, e_L)^T$ & $0$ & $1$ & $-1$ & $0$ & $-1/2$ \\
$e_R$               & $0$ & $1$ & $-1$ & $-1/2$ & $-1$ \\
$\nu_R$             & $0$ & $1$ & $-1$ & $+1/2$ & $0$ \\
\bottomrule
\end{tabular}
\end{center}

\noindent
The $T_{3R}$ column gives the right-handed weak isospin values in the full
$SU(2)_R$ theory; these are used to compute the $Y$ values via
$Y = T_{3R} + (B{-}L)/2$.
The hypercharge values so obtained are group-theoretic properties of the
representations, fixed before the OP-S4 decoupling.
Applying OP-S4 (assumption~(A)) decouples the \emph{gauge coupling} to $SU(2)_R$
but does not change the $Y$ eigenvalues, since $Y$ is a generator of the
commuting $U(1)_Y$ factor.
All six values match the Standard Model hypercharge table.
\end{proof}

\begin{remark}[Consistency check via Gell-Mann--Nishijima]
\label{rem:gm_nishijima}
The electric charge formula $Q = T_{3L} + Y$ reproduces the SM charges:
$Q(u_L) = 1/2 + 1/6 = 2/3$, $Q(d_L) = -1/2 + 1/6 = -1/3$,
$Q(e_L) = -1/2 - 1/2 = -1$, $Q(\nu_L) = 1/2 - 1/2 = 0$.
This is a consistency check, not a new derivation.
\end{remark}

\section{Sub-Gap C2-i: Baryon Number $B_q=1/3$ from SU(3) Colour Structure}
\label{sec:c2i}
%─────────────────────────────────────────────────────────────────────────────

Assumption~(C) in Lemma~\ref{lem:hypercharge_formula} asserts $B=1/3$ per quark.
In the UBT SU(3) sector, quarks sit in the fundamental representation
$\mathbf{3}$ of $SU(3)_c$, generated by $\mathbb{Z}_2{\times}\mathbb{Z}_2{\times}\mathbb{Z}_2$
involutions on $\mathbb{C}\otimes\mathbb{H}$
(\texttt{canonical/su3\_derivation/su3\_from\_involutions.tex}).

\begin{lemma}[Baryon number $B_q=1/3$ from SU(3) colour-singlet constraint,
  \textbf{[L1 conditional on SU(3) colour structure from UBT]}]
\label{lem:Bq_from_su3}
Assume:
\begin{enumerate}
  \item[(i)] $SU(3)_c$ is established in UBT with quarks in the fundamental
             representation $\mathbf{3}$
             (\texttt{canonical/interactions/sm\_gauge.tex §G.B}, [L1]).
  \item[(ii)] Physical colour-neutral states (baryons) are antisymmetric three-quark
              singlets $\varepsilon^{abc}\,q_a q_b q_c$ — the unique colour-singlet
              formed from three fundamentals of $SU(3)_c$ ([L0], standard representation theory).
  \item[(iii)] Baryons carry baryon number $B=1$ by definition; anti-baryons carry $B=-1$.
\end{enumerate}
Then every quark in $\mathbf{3}$ carries baryon number
\begin{equation}
  \boxed{B_q \;=\; \tfrac{1}{3},}
  \label{eq:Bq_one_third}
\end{equation}
as a global $U(1)_B$ charge additive across the three colour components.
\end{lemma}

\begin{proof}
A baryon consists of exactly three quarks (one from each colour sector $a=1,2,3$)
combined via the $\varepsilon^{abc}$ tensor into a colour-singlet.
Baryon number $B$ is additive:
\[
  B_{\mathrm{baryon}} = B_{q_1} + B_{q_2} + B_{q_3} = 3\,B_q.
\]
Setting $B_{\mathrm{baryon}} = 1$ (assumption~(iii)) gives $B_q = 1/3$.
This assignment is $SU(3)_c$-invariant because $B$ commutes with all $SU(3)$
generators: $[U(1)_B,\, SU(3)_c] = 0$ (standard result, [STD]).
The three colour components of a single quark carry equal $B_q=1/3$ due to
the $\mathbb{Z}_2{\times}\mathbb{Z}_2{\times}\mathbb{Z}_2$ involution symmetry
among the three sectors, which acts transitively on the colour triplet.
\end{proof}

\begin{remark}[Status of assumption (iii)]
\label{rem:Bbaryon_convention}
The assignment $B_{\mathrm{baryon}}=1$ in assumption~(iii) is a normalisation
convention, not an independently derived quantity.  Within UBT, baryons are
identified as the topological three-quark bound states that emerge from the
$\varepsilon^{abc}$ colour-singlet constraint; the assignment $B=1$ for this
state is the UBT \emph{definition} of the baryon-number unit.
The Lemma therefore derives $B_q=1/3$ from the SU(3) representation structure
given this conventional normalisation.  The result is [L1 conditional on the
SU(3) colour structure of UBT (assumption (i)–(ii))]; the normalisation
of $B$ (assumption (iii)) is a definition, not a further gap.
\end{remark}

\begin{remark}[Residual open question: $U(1)_B$ from $\psi$-winding]
\label{rem:c2ii_psi_winding}
Lemma~\ref{lem:Bq_from_su3} derives $B_q=1/3$ from the SU(3) colour-singlet
constraint without appealing to $\psi$-winding topology.
A separate (and deeper) question is whether $U(1)_B$ itself arises as a
$\psi$-winding topological symmetry, analogously to how the three generations
arise from winding numbers $n=1,2,3$.

In the current UBT framework, $U(1)_Y$ is identified with the right scalar
phase on $S^1_\psi$ (\texttt{canonical/interactions/sm\_gauge.tex}), and $U(1)_B$
is a \emph{separate} global phase symmetry.  The identification of $U(1)_B$
with a $\psi$-winding sector has not been established and constitutes a
named sub-problem:

\textbf{Sub-gap C2-ii [OPEN/MC]:} Derive $U(1)_B$ as a $\psi$-winding
topological symmetry from $S[\Theta]$, showing that the fractional winding
$n_q = 1/3$ for quarks (versus $n_\ell = 1$ for leptons) follows from
the $\mathbb{Z}_2{\times}\mathbb{Z}_2{\times}\mathbb{Z}_2$ involution
structure of SU(3).

\medskip
\noindent\textbf{Approach via $\mathbb{Z}_2^3$ involution structure (v57 attempt).}
The three $\mathbb{Z}_2$ involutions $\tau_1,\tau_2,\tau_3$ on
$\mathcal{B}=\mathbb{C}\otimes\mathbb{H}$ (defined as $\tau_k(\Theta)=\sigma_k\Theta\sigma_k$
with Pauli matrices $\sigma_k$) generate the colour sector and satisfy
$\tau_1\circ\tau_2\circ\tau_3=\mathrm{id}$,
giving a $\mathbb{Z}_3$ cyclic structure
(\texttt{canonical/interactions/colour\_charge\_lattice.tex}, Lem.~p=3).
Each quark colour eigenstate $(|r\rangle,|g\rangle,|b\rangle)$ is associated
with exactly one involution sector.

The candidate Lemma for C2-ii is the following:

\begin{lemma}[$n_q=1/3$ from $\mathbb{Z}_3$ cyclic involution symmetry (Motivated Candidate)]
\label{lem:c2ii_candidate}
Assume:
\begin{enumerate}
  \item[(i)] $\psi$-winding is additive over involution sectors, i.e.\ the
    total winding of a colour-singlet (baryon) equals the sum of windings of
    its constituent colour components: $n_{\mathrm{baryon}}=n_r+n_g+n_b$.
  \item[(ii)] The $\mathbb{Z}_3$ cyclic structure $(\tau_1\circ\tau_2\circ\tau_3=\mathrm{id})$
    implies that the three colour sectors are equivalent, so
    $n_r=n_g=n_b=:n_q$.
  \item[(iii)] A colour-singlet baryon and a lepton carry the same total
    $\psi$-winding: $n_{\mathrm{baryon}}=n_\ell=1$.
\end{enumerate}
Then $n_q = 1/3$.
\end{lemma}

\begin{proof}
From (i) and (ii): $n_{\mathrm{baryon}}=3n_q$.
From (iii): $3n_q=1$, hence $n_q=1/3$.
\end{proof}

\noindent
The algebraic step is immediate; the content is entirely in assumptions (i)--(iii).
Assumption (i) is standard (winding numbers are additive for product states).
Assumption (ii) follows from the $\mathbb{Z}_3$ cyclic symmetry of the involutions.
\textbf{Status: [MC].}  Lemma~\ref{lem:c2ii_candidate} is a Motivated Candidate because
assumption (iii) has not been derived from $S[\Theta]$ (see C2-iii below).

\medskip
\noindent\textbf{Sub-gap C2-iii (new, v57):}
Assumption (iii) of Lemma~\ref{lem:c2ii_candidate} is not derived from $S[\Theta]$.
Specifically:
\begin{itemize}
  \item In the current UBT formalism, $S^1_\psi$ lies in the imaginary-time
    direction of $\tau=t+i\psi$ (complex time), while SU(3) colour acts on the
    rank-3 internal fiber $\mathbb{C}^3\subset\Theta$.  These are orthogonal
    degrees of freedom.
  \item There is no axiom or derived result in $S[\Theta]$ that couples the
    $\psi$-winding number to the colour representation content.
  \item Establishing assumption (iii) requires deriving a geometric or
    topological coupling: $S^1_\psi \leftrightarrow \mathbb{Z}_3$ colour cycle.
\end{itemize}

\textbf{C2-iii [OPEN/MC]:} Derive the coupling $n_{\mathrm{baryon}}=n_\ell=1$
from $S[\Theta]$, i.e.\ show that the $\psi$-winding budget of a colour-singlet
state equals that of a lepton.  This requires an explicit geometric or
topological map between the imaginary-time circle $S^1_\psi$ and the
$\mathbb{Z}_3$ cyclic structure of the colour involutions.
Until C2-iii is closed, Lemma~\ref{lem:c2ii_candidate} remains [MC] and
C2-ii remains [OPEN/MC].

\end{remark}

\begin{remark}[C2-iii resolution attempts (v58–v59); C2-iv formal NO-GO]
\label{rem:c2iii_attempts}
Three approaches were examined to derive assumption~(iii) of
Lemma~\ref{lem:c2ii_candidate} from $S[\Theta]$.  None succeeds;
the fundamental obstacle is recorded as a formal NO-GO for C2-iv below.

\medskip
\noindent\textbf{Step~1 (Lemma for period $p=3$ route, \texttt{colour\_charge\_lattice.tex}).}
The Lemma for period $p=3$ in
\texttt{canonical/interactions/colour\_charge\_lattice.tex}
establishes $\tau_1\circ\tau_2\circ\tau_3=\mathrm{id}$ [L0], giving a
$\mathbb{Z}_3$ cyclic structure with period $p=3$.
This $\mathbb{Z}_3$ acts on the colour fiber $\mathbb{C}^3\subset\Theta$
(internal colour index).
Colour-singlet projectors $\varepsilon^{abc}q_aq_bq_c$ are invariant under
this $\mathbb{Z}_3$ colour rotation.
However, $\psi$-winding is an eigenvalue of $\psi$-translation
$T_\psi\colon\Theta\mapsto e^{in\psi}\Theta$, which acts on the complex-time
fiber $\tau=t+i\psi$.
The colour $\mathbb{Z}_3$ acts on the colour index $a\in\{1,2,3\}$ and
commutes with $\psi$-translation; no term in $S[\Theta]$ couples them.
\textbf{Conclusion:} Lemma for period $p=3$ does not imply that colour-singlet
projectors carry a definite $\psi$-winding number.

\medskip
\noindent\textbf{Step~2 (topological homeomorphism route).}
One might seek a natural $\mathbb{Z}_3$ action on $S^1_\psi$ such that
$S^1_\psi/\mathbb{Z}_3\cong[0,2\pi/3]$, hoping that winding on the
quotient corresponds to fractional winding $n_q=1/3$ on $S^1_\psi$.
However, for $\mathbb{Z}_3$ to act on $S^1_\psi$ by rotation
$\psi\mapsto\psi+2\pi/3$, the colour-sector $\mathbb{Z}_3$ (acting on
$\mathbb{C}^3$) must be identified with a translation symmetry of the
$\psi$-circle.
In $S[\Theta]$, the minimal-coupling prescription
$D_\mu\Theta=(\partial_\mu-igA_\mu)\Theta$ acts on spacetime indices
$\mu=0,\ldots,3$ and does not include a $\psi$-direction coupling for
$SU(3)_c$.
The homeomorphism route is therefore circular: it requires assuming
the coupling that C2-iii is meant to derive.
\textbf{Conclusion:} No natural $\mathbb{Z}_3$ action on $S^1_\psi$
is generated by the colour $\mathbb{Z}_3$ in $S[\Theta]$.

\medskip
\noindent\textbf{Step~3 / Approach~(b) (v59): $\mathbb{Z}_3$ from
$\tau_1\circ\tau_2\circ\tau_3$ acting on $S^1_\psi$ via $\tau=t+i\psi$.}

Since $\tau_1\circ\tau_2\circ\tau_3=\mathrm{id}$ [L0] is a $\mathbb{Z}_3$
cyclic structure from the SU(3) involutions, we probe whether it could
induce a natural rotation $\psi\mapsto\psi+2\pi/3$ on $S^1_\psi$ via the
complex structure $\tau=t+i\psi$.

The UBT involutions are defined as fiber automorphisms:
\[
  \tau_k(\Theta)(q,\tau) \;=\; \sigma_k\,\Theta(q,\tau)\,\sigma_k,
\]
acting pointwise on the \emph{value} $\Theta$ at each $(q,\tau)$, while
leaving the base-space coordinate $\tau=t+i\psi$ unchanged.

For a $\mathbb{Z}_3$ rotation $\psi\mapsto\psi+2\pi/3$ to be generated, one
would need a base-space map $\widetilde{\tau}_k$ such that
\[
  \widetilde{\tau}_k(\Theta)(q,t+i\psi)
  \;=\;
  \Theta\!\left(q,\,t+i\!\left(\psi+\tfrac{2\pi}{3}\right)\right),
\]
i.e.\ a shift of the imaginary-time coordinate.  No such definition is part
of the UBT involution structure: $\tau_k$ are algebra-fiber automorphisms and
carry no action on the argument $\tau\in\mathbb{C}$.

Moreover, $\tau_1\circ\tau_2\circ\tau_3=\mathrm{id}$ means the composition
is the identity on field \emph{values}.  An identity map on $\Theta(q,\tau)$
cannot encode a non-trivial shift of the coordinate $\psi$.  Therefore no
combination of $\{\tau_1,\tau_2,\tau_3\}$ generates $\psi\mapsto\psi+2\pi/3$.

\textbf{Conclusion:} Approach~(b) is a NO-GO.  The $\mathbb{Z}_3$ cycle from
$\tau_1\circ\tau_2\circ\tau_3=\mathrm{id}$ is a fiber-direction automorphism
and has no action on the base-space $\psi$-coordinate.

\medskip

\noindent\textbf{Sub-gap C2-iv [FORMAL NO-GO] (v59) — Fiber Decoupling in $S[\Theta]$:}
Steps~1, 2, and~3 share a common root cause, now recorded as a formal NO-GO:

In $S[\Theta]$, the imaginary-time circle $S^1_\psi$ and the colour
fiber $\mathbb{C}^3$ are \emph{algebraically decoupled} degrees of freedom.
\begin{itemize}
  \item[(a)] $S^1_\psi$ is the compactification of the imaginary part of
    $\tau=t+i\psi$; its winding group is $\pi_1(S^1_\psi)=\mathbb{Z}$.
  \item[(b)] The colour fiber $\mathbb{C}^3$ carries $SU(3)_c$, generated
    by $\mathbb{Z}_2^3$ involutions on $\mathbb{C}\otimes\mathbb{H}$;
    its relevant discrete symmetry is $\mathbb{Z}_3$.
  \item[(c)] The current UBT action contains no term that identifies
    $\psi$-winding eigenvalues with colour-representation eigenvalues.
    Adding such a term would require extending $S[\Theta]$ beyond its
    present formulation, and is not implied by any current axiom.
\end{itemize}

Until C2-iv is resolved, Lemma~\ref{lem:c2ii_candidate} remains [MC],
C2-iii and C2-ii both remain [OPEN/MC].
These sub-gaps do \emph{not} block the hypercharge formula or any current paper.
