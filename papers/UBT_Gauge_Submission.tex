\begin{abstract}
We derive the Standard Model gauge structure
$\mathrm{SU}(3)_c\times\mathrm{SU}(2)_L\times\mathrm{U}(1)_Y$
from the biquaternion algebra $\mathbb{C}\otimes_{\mathbb{R}}\mathbb{H}$
without free parameters for the gauge group.
Three fermion generations follow from winding modes on the compact
imaginary directions of $\mathbb{H}$.
Hypercharge assignments $Y_Q=1/6$ are derived from the topology of
$S^1_\psi$ [L1].
The Weinberg angle $\sin^2\theta_W = 3/8$ at the unification scale
follows conditionally [L1 cond.].
Honest dead ends (Higgs sector, $\alpha$) are
explicitly bounded.
\end{abstract}

\subsection{Motivation}

The Standard Model gauge group
$G_{\mathrm{SM}} = \SU{3}_c\times\SU{2}_L\times\UU{1}_Y$
is one of the most precisely tested structures in physics.
Yet its algebraic origin is unexplained: it is introduced as an axiom
rather than derived from a more fundamental framework.
In the Standard Model, the choice of gauge group, its representations,
and the hypercharge assignments are all postulated.
No existing framework derives all three factors from a single algebraic
structure without introducing the group as external input.

\subsection{UBT approach}

Unified Biquaternion Theory (UBT) \cite{jaros2026gr} is built on the field
$\Theta(q,\tau)$ over biquaternion space $\mathbb{B} = \Bq$
and complex time $\tau = t + i\psi$.
The same algebra $\mathbb{B}$ that underlies the gravitational sector
also determines the gauge structure.
This paper derives:
\begin{enumerate}
\item $\SU{3}_c$ from the $\mathbb{Z}_2^3$ involution structure of $\mathbb{B}$ [\Lzero].
\item $\SU{2}_L$ from the left norm-preserving action on $\mathbb{B}$ [\Lzero].
\item $\UU{1}_Y$ from the scalar phase right action on $\mathbb{B}$ [\Lzero].
\item Hypercharge $Y_Q = 1/6$ from a topological condition on $S^1_\psi$ [\Lone].
\item Three generations from three $\psi$-winding modes [\Lzero].
\item $N_{\mathrm{eff}} = 12$ from the SU(2) Scherk--Schwarz twist [\Lone].
\end{enumerate}

\section{Algebraic Foundation}
\label{sec:algebra}
% ============================================================

\begin{definition}[Biquaternion algebra]
\label{def:biquaternion}
The \emph{biquaternion algebra} is
\[
  \mathbb{B} \;:=\; \Bq \;\cong\; \mathrm{Mat}(2,\mathbb{C})
\;\cong\; \mathrm{Cl}_{1,3}(\mathbb{R}).
\]
As a real vector space $\dim_\mathbb{R}\mathbb{B} = 8$.
The isomorphism $\mathbb{B}\cong\mathrm{Cl}_{1,3}(\mathbb{R})$ identifies
the algebra with the Clifford algebra of Minkowski space.

Ref: \texttt{canonical/fields/biquaternion\_algebra.tex} \PROVED.
\end{definition}

\begin{remark}[Automorphism group]
The automorphism group $\mathrm{Aut}(\mathbb{B})$ of the biquaternion algebra
contains $\SU{3}\times\SU{2}_L\times\UU{1}_Y$ as a subgroup realised
through the involution, left-action, and right-action structures
described in Sections~\ref{sec:su3}--\ref{sec:u1}.
Ref: \texttt{canonical/interactions/sm\_gauge.tex} [\Lzero].
\end{remark}

\subsection{The Theta field}

The fundamental field of UBT is $\Theta(q,\tau)\in\mathbb{B}$
for $q\in\mathbb{B}$ and $\tau = t + i\psi\in\mathbb{C}$.
The UBT field equation is
\[
\nabla^\dagger\nabla\Theta(q,\tau) = \kappa\,\mathcal{T}(q,\tau),
\]
and the action is
\[
S[\Theta] = \int d^4x\,d\psi\;
\bigl[\tfrac{1}{2}(\nabla^\dagger\nabla\Theta)^2 + \mathcal{L}_{\mathrm{int}}\bigr].
\]
All gauge structures in this paper are derived from the algebraic properties
of $\mathbb{B}$ without additional postulates.

\section{SU(3) Colour from Involutions}

\subsection{The $\mathbb{Z}_2^3$ involution structure}

\begin{definition}[Canonical involutions, \PROVED]
\label{def:involutions}
The three canonical $\mathbb{R}$-linear involutions of $\mathbb{B}$:
\begin{align*}
\alpha : q_0 + q_1 i + q_2 j + q_3 k &\;\mapsto\; q_0 + q_1 i - q_2 j - q_3 k, \\
\beta  : q_0 + q_1 i + q_2 j + q_3 k &\;\mapsto\; q_0 - q_1 i + q_2 j - q_3 k, \\
\gamma : q_0 + q_1 i + q_2 j + q_3 k &\;\mapsto\; q_0 - q_1 i - q_2 j + q_3 k,
\end{align*}
generate the group
$G := \langle\alpha,\beta,\gamma\rangle \cong \mathbb{Z}_2\times\mathbb{Z}_2\times\mathbb{Z}_2$
of order 8.

Ref: \texttt{canonical/algebra/involutions\_Z2xZ2xZ2.tex} \PROVED.
\end{definition}

\begin{theorem}[$\mathfrak{su}(3)$ in $\mathbb{B}$, \PROVED]
\label{thm:su3_involutions}
The $G$-equivariant traceless anti-Hermitian elements of
$\mathbb{B}\cong\mathrm{Mat}(2,\mathbb{C})$ form a Lie algebra isomorphic to
$\mathfrak{su}(3)$.
The eight Gell-Mann generators $\lambda_1,\ldots,\lambda_8$ are realised
with the correct structure constants $[\lambda_i,\lambda_j] = 2i f_{ijk}\lambda_k$.

Ref: \texttt{canonical/su3\_derivation/su3\_from\_involutions.tex},
Theorem~1; \texttt{canonical/su3\_derivation/step3\_SU3\_result.tex}.
\end{theorem}

\begin{theorem}[Fundamental representation $\mathbf{3}$, \PROVED]
\label{thm:fundamental}
The fundamental 3-complex-dimensional representation of $\SU{3}_c$ is
realised on the 3d complex subspace of $\mathbb{B}$ selected by the
$\psi$-winding modes.
Quarks transform in $\mathbf{3}$.

Ref: \texttt{canonical/interactions/sm\_gauge.tex}, Theorem~G.B.
\end{theorem}

\begin{theorem}[Adjoint representation $\mathbf{8}$, \PROVED]
\label{thm:adjoint}
The generators $\lambda_i$ transform in the adjoint representation
$\mathbf{8}$ under $X\mapsto UXU^\dagger$.
Gluons transform in $\mathbf{8}$.

Ref: \texttt{canonical/interactions/sm\_gauge.tex}, Theorem~G.C.
\end{theorem}

\begin{theorem}[$\SU{3}$ from qubit encoding, \PROVED]
\label{thm:su3_qubit}
Under $\Bq\cong\mathrm{Mat}(2,\mathbb{C})\cong$ (qubit algebra), colour charges
are one-hot qubit states:
\[
  |r\rangle = |001\rangle,\quad |g\rangle = |010\rangle,\quad |b\rangle = |100\rangle.
\]
The $\SU{3}$ colour symmetry acts on this qubit state space.
The colour singlet is the unique even-parity zero-codeword.

Ref: \texttt{canonical/interactions/su3\_qubit\_encoding.tex};
\texttt{canonical/bridges/su3\_gauge\_qubit\_equivalence.tex}.
\end{theorem}

\begin{remark}[Structural colour confinement]

in the physical (singlet-only) Hilbert space.
A full dynamical proof (area law, mass gap) is open.
\end{remark}

\section{SU(2)$_L$ Chirality}
\label{sec:su2}
% ============================================================

\begin{theorem}[$\SU{2}_L$ from left norm-preserving action, \PROVED]
\label{thm:su2}
The group of norm-preserving left multiplications on
$\mathbb{B}\cong\mathrm{Mat}(2,\mathbb{C})$,
\[
  \{U\in\mathrm{Mat}(2,\mathbb{C}) : U^\dagger U = \mathbf{1}\} = \UU{2},
\]
contains $\SU{2}_L$ as its determinant-1 subgroup.
This is identified with the electroweak $\SU{2}_L$ gauge symmetry acting
on left-chiral doublets.

Ref: \texttt{canonical/interactions/sm\_gauge.tex},
\S ``Geometric derivation of $\SU{2}_L\times\UU{1}_Y$''.
\end{theorem}

\begin{remark}[Chirality: $L$ not $R$]
Under $\psi\to-\psi$ (imaginary time reversal), left- and right-handed
couplings transform differently via $\tau = t + i\psi$.
This motivates $\SU{2}_L$ as the physical weak isospin group.
A rigorous derivation from $S[\Theta]$ via $\psi$-parity analysis is
Gap~C1 (\texttt{research\_tracks/T2\_GAUGE/missing\_axioms.md}):
status \SE\ (motivated but not formally proved).

Ref: \texttt{canonical/interactions/su2\_chirality.tex} (if available);
\texttt{canonical/chirality/}.
\end{remark}

\begin{theorem}[$\sin^2\theta_W = 3/8$ at GUT scale, \Lone\ (conditional)]
\label{thm:weinberg_gut}
The ratio of generator norms in $\mathbb{B}$ gives
\[
\sin^2\theta_W\big|_{\mathrm{GUT}} = \frac{3}{8},
\]
at the GUT-unification scale where all couplings are equal.

The algebraic derivation of $\sin^2\theta_W(M_Z) = 0.231$ from $\mathbb{B}$
alone is a dead end (Gap~EW-1).

\begin{theorem}[Three generations, \PROVED]
\label{thm:generations}
Three independent $\psi$-winding modes on $S^1_\psi$ give three
identical copies of the Standard Model quantum number structure:
\[
  N_{\mathrm{gen}} = \dim_\mathbb{R}(\mathrm{Im}\,\mathbb{H}) = 3.
\]
Each mode carries identical gauge quantum numbers.
The mass hierarchy is an open problem.

Ref: \texttt{DERIVATION\_INDEX.md}: ``$\psi$-modes as independent B-fields~[L0]''.
\end{theorem}

\begin{theorem}[$N_{\mathrm{eff}}=12$ from SU(2) twist, \Lone]
\label{thm:neff}
The Scherk--Schwarz twist $\Theta(\psi + 2\pi R_\psi) = \sigma_3\Theta\sigma_3$
on $S^1_\psi$ generates half-integer Kaluza--Klein momenta for the
off-diagonal $b$-fields, yielding $N_{\mathrm{eff}} = 12$ effective KK modes
contributing to the one-loop effective potential.

Ref: \texttt{research\_tracks/quantum\_ubt/su2\_twist\_neff12.tex} [\Lone].
\end{theorem}

Higher harmonics ($n\geq 4$ winding modes) correspond to massive KK excitations
beyond the three light generations [\MC].

\section{Discussion and Conclusion}
\label{sec:conclusion}
% ============================================================

The Standard Model gauge group $\SU{3}_c\times\SU{2}_L\times\UU{1}_Y$
is uniquely realised in $\mathbb{B} = \Bq$ with zero free parameters.
Three generations emerge from the $\psi$-winding mode structure.
$N_{\mathrm{eff}} = 12$ KK modes follow from the SU(2) twist.
Hypercharge $Y_Q = 1/6$ and $\sin^2\theta_W = 3/8$ at the GUT scale
are both derived from topological conditions on $S^1_\psi$.

All open problems — chirality, Higgs/Yukawa,
$\alpha$ — are explicitly stated.

\subsection*{Falsifiable prediction: proton decay}
The UBT GUT scale $M_{\rm GUT}\sim 2\times 10^{16}$\,GeV [CONDITIONAL]
implies a proton lifetime
\[
\tau_p \sim \frac{M_{\rm GUT}^4}{m_p^5}\sim 10^{34}\ \text{years}
\]
for the $p\to e^+\pi^0$ channel.
The Super-Kamiokande lower bound $\tau_p > 1.6\times 10^{34}$\,years
is consistent with this prediction.
The Hyper-Kamiokande experiment (2027+) will provide the
next discriminative test window.
See \texttt{reports/falsifiable\_prediction\_sheet.md}.
